#!/usr/bin/env python3
# ===============================================================
# STUR 5D Helical Manifold Lab – v4.0 (PHYSICAL MODE DEFAULT)
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, math
plt.style.use("dark_background")

MODE = "physical"  # default as requested

def is_extreme(): return MODE.lower().startswith("ext")
def is_physical(): return MODE.lower().startswith("phys")

# Parameters
c=1.0
TAU_MAX = 8.0*np.pi
N_SAMPLES = 1000
R_HELIX = 1.0
PITCH_Z = 0.1
CHI0 = 1.0
K_CHI = 0.05
ALPHA0 = 1.0
OMEGA_A = 0.25
EPS_CHI_PHYS = 0.1
EPS_CHI_EXT = 0.6
OMEGA4_GAMMA = 0.3
LAMBDA_T_BASE = 0.2
COUP_TCHI_BASE = 0.15
COUP_ZCHI_BASE = 0.10
OMEGA_COUP = 0.40

def current_eps_chi(): return EPS_CHI_EXT if is_extreme() else EPS_CHI_PHYS
def current_couplings():
    return (1.0,0.6,0.4) if is_extreme() else (LAMBDA_T_BASE,COUP_TCHI_BASE,COUP_ZCHI_BASE)

def alpha_of_tau_chi(tau,chi):
    eps=current_eps_chi()
    base = ALPHA0*np.cos(OMEGA_A*tau)
    if is_physical():
        return base*(1+eps*np.sin(0.7*chi))
    core=1+eps*np.tanh(chi)*np.sin(0.5*tau)
    return base*core**3 + 0.03*np.sign(base)

def chi_of_tau(tau,b=+1):
    return b*(CHI0+K_CHI*tau)

def helix_xyz_of_tau(tau,chi):
    if is_physical():
        return (R_HELIX*np.cos(tau),R_HELIX*np.sin(tau),PITCH_Z*tau)
    a=np.tanh(alpha_of_tau_chi(tau,chi))
    R=R_HELIX*(1+0.45*a*np.tanh(chi))
    phi=tau+0.6*np.tanh(chi)*np.sin(0.4*tau)
    z=PITCH_Z*tau*(1+0.8*a*np.exp(-0.2*(tau-0.5*TAU_MAX)**2/(0.1*TAU_MAX**2)))
    return (R*np.cos(phi),R*np.sin(phi),z)

def compute_ds2(S):
    tau=S["tau"]
    t=tau
    x=S["x"];y=S["y"];z=S["z"];chi=S["chi"]
    lamT,cT,cZ=current_couplings()
    g_tt=-(c**2)*(1+lamT*np.sin(OMEGA_COUP*tau)*np.tanh(chi)**2)
    g_chi_chi=alpha_of_tau_chi(tau,chi)
    g_t_chi=cT*np.sin(OMEGA_COUP*tau)*np.tanh(chi)
    g_z_chi=cZ*np.cos(OMEGA_COUP*tau)*np.tanh(chi)
    dt=tau
    dx=np.gradient(x,tau);dy=np.gradient(y,tau)
    dz=np.gradient(z,tau);dchi=np.gradient(chi,tau)
    ds2=g_tt+dx**2+dy**2+dz**2+g_chi_chi*dchi**2 \
        +2*g_t_chi*dchi+2*g_z_chi*dz*dchi
    dτ=np.gradient(tau);return ds2*dτ

def build_strand(b):
    tau=np.linspace(0,TAU_MAX,N_SAMPLES)
    chi=chi_of_tau(tau,b)
    x,y,z=helix_xyz_of_tau(tau,chi)
    return dict(branch=b,tau=tau,x=x,y=y,z=z,chi=chi,
                alpha=alpha_of_tau_chi(tau,chi))

def plot_all():
    sp=build_strand(+1); sm=build_strand(-1)

    fig=plt.figure(figsize=(9,7));ax=fig.add_subplot(111,projection="3d")
    ax.scatter(sp["x"],sp["y"],sp["z"],c=sp["chi"],s=6)
    ax.scatter(sm["x"],sm["y"],sm["z"],c=sm["chi"],s=6,marker="^")
    ax.set_title("3D Projection – "+MODE.upper());plt.show()

    plt.figure(figsize=(6,4))
    plt.plot(sp["tau"],sp["chi"],label="chi+")
    plt.plot(sm["tau"],sm["chi"],label="chi-")
    plt.legend();plt.title("chi(τ)");plt.show()

    ds2=compute_ds2(sp);Sτ=np.cumsum(np.sqrt(np.abs(ds2))*np.sign(ds2))
    plt.figure(figsize=(6,4))
    plt.plot(sp["tau"],Sτ);plt.title("Proper Interval S(τ)");plt.show()

    summary=json.dumps({
        "model":"STUR_5D_v4",
        "mode":MODE,
        "chi_range":[float(np.min(sp["chi"])),float(np.max(sp["chi"]))],
    },indent=2)
    print(summary)

if __name__=="__main__":
    plot_all()
