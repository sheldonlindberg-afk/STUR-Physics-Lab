def friedmann_E(a):
    return np.sqrt(Omega_r * a**(-4) + Omega_m * a**(-3) + Omega_L)

def build_cosmo_background():
    a_grid = np.logspace(np.log10(A_MIN), 0.0, N_COSMO)
    E = friedmann_E(a_grid)

    integrand = 1.0 / (a_grid * E + 1e-30)
    t_grid = np.zeros_like(a_grid)
    t_grid[1:] = np.cumsum(
        0.5 * (integrand[1:] + integrand[:-1]) * (a_grid[1:] - a_grid[:-1])
    )
    u_t = t_grid / t_grid[-1]   

    dE_du = np.gradient(E, u_t)
    R_FRW = 6.0 * (dE_du + 2.0 * E**2)

    rho_r = Omega_r * a_grid**(-4)
    rho_m = Omega_m * a_grid**(-3)
    rho_L = Omega_L * np.ones_like(a_grid)
    rho_tot = rho_r + rho_m + rho_L

    Om = rho_m / rho_tot
    OL = rho_L / rho_tot

    frac_b   = Omega_b / Omega_m
    frac_cdm = Omega_cdm / Omega_m
    Ob = frac_b   * Om
    Oc = frac_cdm * Om

    mu = 1.0 / a_grid
    mu /= mu[-1]
    log10_mu = np.log10(mu)

    lnE = np.log(E)
    lnA = np.log(a_grid)
    dlnE_dlnA = np.gradient(lnE, lnA)
    q_grid = -1.0 - dlnE_dlnA

    lnE2 = np.log(E**2)
    dlnE2_dlnA = np.gradient(lnE2, lnA)
    w_eff_grid = -1.0 - (1.0/3.0) * dlnE2_dlnA

    g = 1.0 / (a_grid**3 * E**3 + 1e-30)
    G_int = np.zeros_like(a_grid)
    G_int[1:] = np.cumsum(
        0.5 * (g[1:] + g[:-1]) * (a_grid[1:] - a_grid[:-1])
    )
    D_raw = E * G_int
    D = D_raw / (D_raw[-1] + 1e-30)

    return (
        a_grid, u_t, E, R_FRW,
        Or, Om, OL, Ob, Oc,
        log10_mu, q_grid, w_eff_grid, D

def solve_geodesic_deviation_full(tau, K_total, epsilons):
    N = len(tau)
    h = tau[1] - tau[0]
    n_eps = len(epsilons)
    eta_all = np.zeros((n_eps, 5, N))
    
    for ei, eps in enumerate(epsilons):
        eta = np.array([0.0, eps, eps, 0.0, eps])
        v   = np.zeros(5)
        eta_all[ei, :, 0] = eta
        for i in range(N - 1):
            K_i = K_total[:, :, i]
            def rhs(ev, vv): return vv, -K_i.dot(ev)
            k1_eta, k1_v = rhs(eta, v)
            k2_eta, k2_v = rhs(eta + 0.5*h*k1_eta, v + 0.5*h*k1_v)
            k3_eta, k3_v = rhs(eta + 0.5*h*k2_eta, v + 0.5*h*k2_v)
            k4_eta, k4_v = rhs(eta + h*k3_eta,     v + h*k3_v)
            eta += (h / 6.0) * (k1_eta + 2*k2_eta + 2*k3_eta + k4_eta)
            v   += (h / 6.0) * (k1_v   + 2*k2_v   + 2*k3_v   + k4_v)
            eta_all[ei, :, i + 1] = eta
    return eta_all, np.linalg.norm(eta_all, axis=1)
