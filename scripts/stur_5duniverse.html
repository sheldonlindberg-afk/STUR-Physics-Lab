def friedmann_E(a):
    """Dimensionless Hubble function E(a) = H(a)/H0."""
    return np.sqrt(Omega_r * a**(-4) + Omega_m * a**(-3) + Omega_L)

def build_cosmo_background():
    """
    Build ΛCDM background on log-spaced a-grid.
    """
    a_grid = np.logspace(np.log10(A_MIN), 0.0, N_COSMO)
    E = friedmann_E(a_grid)

    # cosmic time t/H0^{-1} via dt ∝ da / (a E(a))
    integrand = 1.0 / (a_grid * E + 1e-30)
    t_grid = np.zeros_like(a_grid)
    t_grid[1:] = np.cumsum(
        0.5 * (integrand[1:] + integrand[:-1]) * (a_grid[1:] - a_grid[:-1])
    )
    u_t = t_grid / t_grid[-1]   # normalize so t(a=1) → 1

    # FRW Ricci scalar proxy: R ∝ 6 (dE/du + 2E^2)
    dE_du = np.gradient(E, u_t)
    R_FRW = 6.0 * (dE_du + 2.0 * E**2)

    # energy fractions
    rho_r = Omega_r * a_grid**(-4)
    rho_m = Omega_m * a_grid**(-3)
    rho_L = Omega_L * np.ones_like(a_grid)
    rho_tot = rho_r + rho_m + rho_L
    Or = rho_r / rho_tot
    Om = rho_m / rho_tot
    OL = rho_L / rho_tot

    # matter split
    frac_b   = Omega_b / Omega_m
    frac_cdm = Omega_cdm / Omega_m
    Ob = frac_b   * Om
    Oc = frac_cdm * Om

    # effective energy scale μ ∝ 1/a
    mu = 1.0 / a_grid
    mu /= mu[-1]
    log10_mu = np.log10(mu)

    # deceleration parameter q(a) = -1 - d ln H / d ln a
    lnE = np.log(E)
    lnA = np.log(a_grid)
    dlnE_dlnA = np.gradient(lnE, lnA)
    q_grid = -1.0 - dlnE_dlnA

    # effective equation-of-state:
    #   w_eff = -1 - (1/3) d ln H^2 / d ln a
    lnE2 = np.log(E**2)
    dlnE2_dlnA = np.gradient(lnE2, lnA)
    w_eff_grid = -1.0 - (1.0/3.0) * dlnE2_dlnA

    # linear growth factor D(a):
    #   D(a) ∝ E(a) * ∫^a da' / (a'^3 E^3(a'))
    g = 1.0 / (a_grid**3 * E**3 + 1e-30)
    G_int = np.zeros_like(a_grid)
    G_int[1:] = np.cumsum(
        0.5 * (g[1:] + g[:-1]) * (a_grid[1:] - a_grid[:-1])
    )
    D_raw = E * G_int
    D = D_raw / (D_raw[-1] + 1e-30)

    return (
        a_grid, u_t, E, R_FRW,
        Or, Om, OL, Ob, Oc,
        log10_mu, q_grid, w_eff_grid, D
    )

    K_gauge[2, 2, :] +=  scale_g * OL

    K_gauge[1, 4, :] +=  0.5 * scale_g * OL
    K_gauge[4, 1, :] +=  0.5 * scale_g * OL

    return K_metric, K_torsion, K_gauge

# FULL GEODESIC DEVIATION SOLVER
# ---------------------------------------------------------------
def solve_geodesic_deviation_full(tau, K_total, epsilons):
    """
    Solve η''^a + K^a_c η^c = 0 with full 5x5 K_total(τ).
    Basis: (t, x, y, z, χ) → 5 components.

    epsilons: list of initial perturbation magnitudes ε.
    Initial condition:
      η(0)  = (0, ε, ε, 0, ε)
      η'(0) = 0
    """
    N = len(tau)
    h = tau[1] - tau[0]
    n_eps = len(epsilons)

    eta_all = np.zeros((n_eps, 5, N))
    vel_all = np.zeros((n_eps, 5, N))

    for ei, eps in enumerate(epsilons):
        eta = np.array([0.0, eps, eps, 0.0, eps])
        v   = np.zeros(5)

        eta_all[ei, :, 0] = eta
        vel_all[ei, :, 0] = v

        for i in range(N - 1):
            K_i = K_total[:, :, i]

            def rhs(eta_vec, vel_vec):
                acc = -K_i.dot(eta_vec)
                return vel_vec, acc

            k1_eta, k1_v = rhs(eta, v)
            k2_eta, k2_v = rhs(eta + 0.5 * h * k1_eta, v + 0.5 * h * k1_v)
            k3_eta, k3_v = rhs(eta + 0.5 * h * k2_eta, v + 0.5 * h * k2_v)
            k4_eta, k4_v = rhs(eta + h * k3_eta,     v + h * k3_v)

            eta = eta + (h / 6.0) * (k1_eta + 2 * k2_eta + 2 * k3_eta + k4_eta)
            v   = v   + (h / 6.0) * (k1_v   + 2 * k2_v   + 2 * k3_v   + k4_v)

            eta_all[ei, :, i + 1] = eta
            vel_all[ei, :, i + 1] = v

    mag_all = np.linalg.norm(eta_all, axis=1)
    return eta_all, mag_all

Derivations & Unified Gravity Setup

5D coordinates and worldline. We embed a single closed infinity helix in a 5D manifold with coordinates xa = (t, x, y, z, χ), where χ is the STUR “sideways time” / coherence dimension. The spatial projection is a Gerono lemniscate (∞) with a vertical oscillation: 

x(τ) = R cos τ
y(τ) = (R/2) sin(2τ)
z(τ) = Z_amp sin(2τ)

Three χ-strands represent baryons, cold dark matter, and dark energy: 

χ_b(τ)  =  + (χ₀ + kχ sin τ)
χ_dm(τ) =  - (χ₀ + 1.2 kχ sin(τ + 0.4))
χ_de(τ) =  0.3 + 0.15 sin(2τ)

The reference “observer” worldline is the χ-neutral average: 

χ_ref(τ) = [χ_b(τ) + χ_dm(τ) + χ_de(τ)] / 3.

ΛCDM background. The expansion is a flat ΛCDM cosmology with Planck-like parameters: H₀ = 67.4 km/s/Mpc, Ωm = 0.315, ΩΛ = 0.685, Ωr = 9.2×10⁻⁵, split into baryons and CDM. The dimensionless Hubble function is 

E(a) = H(a)/H₀ = √(Ω_r a⁻⁴ + Ω_m a⁻³ + Ω_Λ).

We integrate the Friedmann equation numerically to get cosmic time u = t/t₀ ∈ [0,1], the FRW Ricci scalar proxy R_FRW(u), energy fractions Ω_r(u), Ω_m(u), Ω_Λ(u), and the linear growth factor D(u) for structure formation.

χ-metric scale and 5D line element. Along the helix, the χ-direction is weighted by a breathing factor α(τ, χ): 

α(τ, χ) = α₀ cos(Ω_A τ) · [1 + ε_χ sin τ · tanh χ]

so the effective 5D line element restricted to the worldline has the schematic form 

ds² = -c² A(τ, χ) dt² + dx² + dy² + dz² + α(τ, χ) dχ²
       + (torsion / gauge cross terms).

Unified geodesic deviation. Instead of solving the full 5D Einstein–Cartan equations, we package their effect into a time-dependent tidal operator Kᵃ₍c₎(τ) acting on a deviation vector ηᵃ(τ) along the χ-neutral worldline: 

ηᵃ″(τ) + Kᵃ₍c₎(τ) ηᶜ(τ) = 0,   a,c = 0..4 (t,x,y,z,χ)

The unified tidal matrix is split into three physically motivated pieces:
