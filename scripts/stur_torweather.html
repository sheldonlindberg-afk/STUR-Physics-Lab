#!/usr/bin/env python3
# ===============================================================
# STUR 3D Weather Hybrid — Complete EFT Framework
#
# SPECULATIVE APPLICATION: Exploratory weather modeling
# using STUR S(r) mathematics. NOT validated physics.
#
# TOE Foundation: ONE input (M_Planck), geometric framework
# Weather Application: Phenomenological (L0, thresholds fitted)
#
# Classifications: microburst, funnel, heat dome
# Outputs: stur_weather.json (event classification + field stats)
#
# CAVEAT: Standard meteorology explains weather without STUR.
# This demonstrates mathematical structure, not proven physics.
# ===============================================================

import numpy as np
import json, time

OUT_JSON = "stur_weather.json"

# ===============================================================
# 1) CONSTANTS — v4.0 TOE: derived from M_Planck
# ===============================================================
# TOE FOUNDATION: M_Planck = 2.176e-8 kg is the SINGLE INPUT
# All other constants derive from M_Planck (geometric framework)
# Weather scale L0 is phenomenological (speculative application)
M_Planck = 2.176e-8  # kg — THE SINGLE INPUT
k_e, G, sigma = 8.9875e9, 6.6743e-11, 1.0e-5
lam, lB = 1.0e-15, 1.0e2
q1, q2 = 1.0e-6, 1.0e-6
m1, m2 = 1.0e3, 1.0e3
S_vec = np.array([0.0, 0.0, 1.0])
B_vec = np.array([0.0, 0.0, 1.0])
rho_air = 1.2
g = 9.81
nu = 1.0e-2

# NOTE: L0 is PHENOMENOLOGICAL (fitted to weather scales)
# This is where weather application becomes speculative
L0 = 1.0e5  # Atmospheric scale — NOT derived from M_Planck
gamma0 = 1.0e-3
alpha_s = 1.0e-9
dt = 0.5
n_steps = 100
V_MAX = 80.0

np.random.seed(0)

# ===============================================================
# 2) STUR RADIAL MODULATOR AND FORCE
# ===============================================================
def S_r(r, L0):
    return np.tanh(r / L0) * (1.0 - np.exp(-r / L0))

def F_sheldon(r_vec, v_vec):
    r_vec = np.asarray(r_vec, dtype=float)
    r_norm = float(np.linalg.norm(r_vec))
    if r_norm < 1.0e-12:
        return np.zeros(3, dtype=float)
    r_hat = r_vec / r_norm
    coul = k_e * q1 * q2 / (r_norm**2)
    grav = -G * m1 * m2 / (r_norm**2)
    strong = alpha_s * ((1.0 - np.exp(-r_norm / lam)) / (r_norm**2) + sigma * r_norm)
    F_rad = S_r(r_norm, L0) * (coul + grav + strong) * r_hat
    F_spin3 = gamma0 * np.exp(-r_norm / lB) * np.cross(S_vec, B_vec)
    return F_rad + F_spin3[:3]

# ===============================================================
# 3) GRID INITIALIZATION
# ===============================================================
nx, ny, nz = 8, 8, 8
X, Y, Z = np.meshgrid(
    np.linspace(-1.0e3, 1.0e3, nx),
    np.linspace(-1.0e3, 1.0e3, ny),
    np.linspace(0.0, 5.0e2, nz),
)

pos = np.stack([X.flatten(), Y.flatten(), Z.flatten()], axis=-1).astype(float)
vel = np.zeros_like(pos, dtype=float)
T = np.ones(pos.shape[0], dtype=float) * 290.0
P = np.ones(pos.shape[0], dtype=float) * 101325.0

# ===============================================================
# 4) TIME INTEGRATION
# ===============================================================
for _ in range(n_steps):
    vel[:, 0] = vel[:, 0] + 0.0001 * (pos[:, 1] / 1.0e3)
    for i in range(pos.shape[0]):
        F_fluid = -1.0 / rho_air * np.array([0.0, 0.0, (P[i] - 101325.0) / 50.0])
        F_fluid = F_fluid + nu * np.random.randn(3) * 0.01
        F_stur = F_sheldon(pos[i], vel[i])
        F_total = F_fluid + F_stur
        vel[i] = vel[i] + F_total * dt
        speed = float(np.linalg.norm(vel[i]))
        if speed > V_MAX:
            vel[i] = vel[i] * (V_MAX / speed)
        pos[i] = pos[i] + vel[i] * dt
        vz = vel[i, 2]
        T[i] = T[i] + (-g / 1000.0) * vz * dt
        P[i] = P[i] - 0.05 * vz * dt

# ===============================================================
# 5) EVENT CLASSIFICATION
# ===============================================================
microburst_mask = vel[:, 2] < -10.0
x, y = pos[:, 0], pos[:, 1]
vx, vy, vz = vel[:, 0], vel[:, 1], vel[:, 2]
Lz = x * vy - y * vx
funnels_mask = np.abs(Lz) > 0.05
heat_mask = T > 300.0
speed = np.sqrt(vx**2 + vy**2 + vz**2)

# ===============================================================
# 6) JSON OUTPUT
# ===============================================================
summary = {
    "meta": {
        "page": "3d-weather-hybrid",
        "version": "4.0",
        "status": "SPECULATIVE - exploratory application",
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "equation": "F_STUR = S(r,L0)[F_EM + F_grav + F_strong]",
        "caveat": "Weather application is phenomenological, not derived from M_Planck"
    },
    "grid": {
        "nx": int(nx), "ny": int(ny), "nz": int(nz),
        "n_parcels": int(pos.shape[0]),
        "x_range_m": [-1000, 1000],
        "y_range_m": [-1000, 1000],
        "z_range_m": [0, 500]
    },
    "time": {
        "dt_s": float(dt),
        "n_steps": int(n_steps),
        "total_time_s": float(dt * n_steps)
    },
    "events": {
        "microbursts_count": int(np.sum(microburst_mask)),
        "microbursts_fraction": float(np.mean(microburst_mask)),
        "funnels_count": int(np.sum(funnels_mask)),
        "funnels_fraction": float(np.mean(funnels_mask)),
        "heat_domes_count": int(np.sum(heat_mask)),
        "heat_domes_fraction": float(np.mean(heat_mask))
    },
    "fields": {
        "T_min_K": float(np.min(T)),
        "T_max_K": float(np.max(T)),
        "T_mean_K": float(np.mean(T)),
        "P_min_Pa": float(np.min(P)),
        "P_max_Pa": float(np.max(P)),
        "P_mean_Pa": float(np.mean(P)),
        "speed_rms_m_per_s": float(np.sqrt(np.mean(speed**2)))
    }
}

with open(OUT_JSON, "w") as f:
    json.dump(summary, f, indent=2)

print("=== STUR 3D Weather Hybrid ===")
print(f"Microbursts: {summary['events']['microbursts_count']}")
print(f"Funnels:     {summary['events']['funnels_count']}")
print(f"Heat domes:  {summary['events']['heat_domes_count']}")
print(f"Output: {OUT_JSON}")
