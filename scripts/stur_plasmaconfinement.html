#!/usr/bin/env python3
"""
STUR Plasma Torus — Multi-Particle 4D
==================================================
Tests STUR-modulated Lorentz force in weak-field GR.

Complete EFT Framework: One input (M_Planck), zero free parameters.
Plasma physics emerges: M_Planck → Z₃ helix → SM gauge group → U(1)_EM + XCRM → confinement

Key equation: du^μ/dτ + Γ^μ_αβ u^α u^β = (q/m) S F^μ_ν u^ν
"""

import numpy as np
import matplotlib.pyplot as plt
import json, time

plt.style.use("dark_background")

# ===============================================================
# CONSTANTS & PARAMETERS
# ===============================================================
c   = 2.99792458e8
G   = 6.67430e-11
M_E = 5.9722e24
R_E = 6.371e6             # Earth radius (m)
q   = 1.602176634e-19     # proton charge
m   = 1.67262192369e-27   # proton mass
Bz  = 1.5                 # Tesla
L0  = 1.0e-3              # STUR length = √2 L_X / σ_R [m] — DERIVED from M_Planck
R_major = 2.0e-2          # torus major radius [m]
a_minor = 4.0e-3          # torus minor radius [m]
Npart   = 400
beta0   = 0.12            # v/c
v0      = beta0 * c
n_periods = 5.0
steps_per_period = 1000

# Cyclotron period for timestep
T_c = 2.0*np.pi*m / (abs(q)*Bz)
dt  = T_c / steps_per_period
Nsteps = int(n_periods * steps_per_period)

rng = np.random.default_rng(42)

# ===============================================================
# GRAVITY (weak-field GR, Earth center below lab)
# ===============================================================
def phi_vec(x, y, z):
    """Newtonian potential with Earth center at z = -R_E"""
    rc2 = x*x + y*y + (z + R_E)**2
    rc  = np.sqrt(rc2)
    return -G*M_E / (rc + 1e-18)

def grad_phi_vec(x, y, z):
    """Gradient of potential → gravitational acceleration"""
    rc2 = x*x + y*y + (z + R_E)**2 + 1e-18
    rc  = np.sqrt(rc2)
    fac = G*M_E / (rc2*rc)
    return fac*x, fac*y, fac*(z + R_E)

# ===============================================================
# STUR SCALAR — Derived from M_Planck via Z₃ geometry
# S(r) = tanh(r/L₀) · (1 - exp(-r/L₀))
# Emergence chain: M_Planck → Z₃ helix → SM → U(1)_EM + XCRM → S(r)
# ===============================================================
def S_STUR_vec(x, y, z):
    """STUR modulator with GR-corrected physical radius"""
    Phi   = phi_vec(x, y, z)
    scale = 1.0 - 2.0*Phi/c**2  # weak-field metric factor
    r2p   = scale * (x*x + y*y + z*z)
    rp    = np.sqrt(np.clip(r2p, 0.0, None))
    u     = rp / L0
    return np.tanh(u) * (1.0 - np.exp(-u))

# ===============================================================
# INITIAL CONDITIONS (torus distribution)
# ===============================================================
theta = rng.uniform(0.0, 2.0*np.pi, Npart)
psi   = rng.uniform(0.0, 2.0*np.pi, Npart)
r_minor = a_minor * (0.6 + 0.4*rng.random(Npart))

x = (R_major + r_minor*np.cos(psi)) * np.cos(theta)
y = (R_major + r_minor*np.cos(psi)) * np.sin(theta)
z =  r_minor * np.sin(psi)

# Tangent velocity with pitch
tx, ty = -np.sin(theta), np.cos(theta)
pitch  = 0.15
norm   = np.sqrt(1.0 + pitch*pitch)
vx = v0 * (tx / norm)
vy = v0 * (ty / norm)
vz = v0 * (pitch / norm)

def gamma_of(vx, vy, vz):
    b2 = (vx*vx + vy*vy + vz*vz) / c**2
    return 1.0 / np.sqrt(1.0 - np.clip(b2, 0.0, 0.9999999))

def limit_speed(vx, vy, vz, vmax=0.99*c):
    v2 = vx*vx + vy*vy + vz*vz
    mask = v2 > vmax*vmax
    if np.any(mask):
        scale = np.sqrt((vmax*vmax) / v2[mask])
        vx[mask] *= scale
        vy[mask] *= scale
        vz[mask] *= scale
    return vx, vy, vz

# ===============================================================
# BORIS ROTATION WITH STUR MODULATION
# ===============================================================
def boris_B_STUR(vx, vy, vz, S, dt):
    """Boris rotation gated by STUR scalar S"""
    g   = gamma_of(vx, vy, vz)
    th  = (q*Bz / (g*m)) * S * dt  # STUR modulates rotation angle
    cth = np.cos(th)
    sth = np.sin(th)
    vxn = cth*vx - sth*vy
    vyn = sth*vx + cth*vy
    return vxn, vyn, vz

# ===============================================================
# DIAGNOSTICS
# ===============================================================
S_hist   = np.zeros(Nsteps)
gamma_mu = np.zeros(Nsteps)
uinv_mu  = np.zeros(Nsteps)

# ===============================================================
# MAIN INTEGRATION LOOP
# ===============================================================
for k in range(Nsteps):
    # Gravity half-kick
    dpx, dpy, dpz = grad_phi_vec(x, y, z)
    gx, gy, gz = -dpx, -dpy, -dpz
    vx += 0.5*dt*gx
    vy += 0.5*dt*gy
    vz += 0.5*dt*gz
    vx, vy, vz = limit_speed(vx, vy, vz)

    # Boris rotation with STUR per particle
    S_loc = S_STUR_vec(x, y, z)
    vx, vy, vz = boris_B_STUR(vx, vy, vz, S_loc, dt)
    vx, vy, vz = limit_speed(vx, vy, vz)

    # Gravity second half-kick
    dpx, dpy, dpz = grad_phi_vec(x, y, z)
    gx, gy, gz = -dpx, -dpy, -dpz
    vx += 0.5*dt*gx
    vy += 0.5*dt*gy
    vz += 0.5*dt*gz
    vx, vy, vz = limit_speed(vx, vy, vz)

    # Drift
    x += vx*dt
    y += vy*dt
    z += vz*dt

    # Diagnostics
    g   = gamma_of(vx, vy, vz)
    inv = -(g*c)**2 + (g*vx)**2 + (g*vy)**2 + (g*vz)**2
    S_hist[k]   = float(np.mean(S_loc))
    gamma_mu[k] = float(np.mean(g))
    uinv_mu[k]  = float(np.mean(inv))

# ===============================================================
# FALSIFICATION CHECK
# ===============================================================
inv_error = abs(np.mean(uinv_mu) + c*c) / (c*c)
excl1 = inv_error > 0.01
excl2 = np.std(S_hist) < 1e-6  # No radius dependence

falsified = excl1 or excl2 or excl3

# ===============================================================
# OUTPUT

print(f"STUR: S(r) = tanh(r/L₀)·(1 - exp(-r/L₀)), L₀ = {L0:.1e} m")
