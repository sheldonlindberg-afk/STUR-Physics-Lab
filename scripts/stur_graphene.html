Executable Simulation

Graphene Dirac transport with axion-like X-bundle coupling. Copy and run in any Python 3 environment.

#!/usr/bin/env python3
# ===============================================================
# STUR Graphene-Axion Coupler
#
# Tests X-bundle axion coupling: g_a φ ψ̄ γ⁵ ψ
# Δτ/τ = C (g_a/g₀)^p with C=5e-6, p=0.65
#
# Outputs:
#   - stur_graphene.json  (full results + falsification flag)
#   - stur_graphene.png   (conductivity vs coupling plot)
#
# Falsified if: Scaling deviates from sub-linear prediction
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, time, math

# ===============================================================
# 1) FUNDAMENTAL CONSTANTS
# ===============================================================
hbar = 1.055e-34    # J·s
e_ch = 1.602e-19    # C
kB   = 1.38e-23     # J/K

# ===============================================================
# 2) GRAPHENE PARAMETERS
# ===============================================================
T = 10.0                      # Temperature [K]
n_cm2 = 1.8e12                # Carrier density [cm⁻²]
n = n_cm2 * 1e4               # [m⁻²]
vF = 1.0e6                    # Fermi velocity [m/s]

kF = math.sqrt(math.pi * n)   # Fermi wavevector
E_F = hbar * vF * kF          # Fermi energy [J]
m_eff = E_F / (vF**2)         # Effective mass

tau0 = 1.0e-13                # Baseline scattering time [s]
B = 10.0                      # Magnetic field [T]

OUT_JSON = "stur_graphene.json"
OUT_PNG  = "stur_graphene.png"

# ===============================================================
# 3) CONDUCTIVITY FUNCTIONS
# ===============================================================
def sigma_xx(tau):
    """Longitudinal conductivity"""
    return n * e_ch**2 * tau / m_eff

def sigma_xy(tau):
    """Hall conductivity"""
    omega_c = e_ch * B / m_eff
    denom = 1.0 + (omega_c * tau)**2
    return n * e_ch**2 * tau * omega_c / (m_eff * denom)

sigma_xx0 = sigma_xx(tau0)
sigma_xy0 = sigma_xy(tau0)

# ===============================================================
# 4) AXION COUPLING MODEL
# ===============================================================
g0_ref = 1.0e-12    # Reference coupling
C_ref  = 5.0e-6     # Amplitude
p_ref  = 0.65       # Sub-linear exponent

def delta_tau_frac(g_a):
    """Δτ/τ = C (g_a/g₀)^p"""
    return C_ref * (g_a / g0_ref)**p_ref

def tau_from_g(g_a):
    return tau0 * (1.0 + delta_tau_frac(g_a))

# ===============================================================
# 5) COUPLING SCAN
# ===============================================================
g_vals = np.logspace(-13, -9, 120)
d_tau_frac = np.array([delta_tau_frac(g) for g in g_vals])
sigma_xx_vals = np.array([sigma_xx(tau_from_g(g)) for g in g_vals])
sigma_xy_vals = np.array([sigma_xy(tau_from_g(g)) for g in g_vals])

d_sigma_xx_frac = (sigma_xx_vals - sigma_xx0) / sigma_xx0
d_sigma_xy_frac = (sigma_xy_vals - sigma_xy0) / sigma_xy0

# Test points
g1, g2 = 1.0e-12, 1.0e-10

def eval_point(g):
    dt = delta_tau_frac(g)
    sxx = sigma_xx(tau_from_g(g))
    sxy = sigma_xy(tau_from_g(g))
    return {
        "g_a": g,
        "delta_tau_over_tau": float(dt),
        "delta_sigma_xx_frac": float((sxx - sigma_xx0)/sigma_xx0),
        "delta_sigma_xy_frac": float((sxy - sigma_xy0)/sigma_xy0)
    }

pt1 = eval_point(g1)
pt2 = eval_point(g2)

# ===============================================================
# 6) FALSIFICATION CHECK
# ===============================================================
# Check sub-linear scaling
expected_ratio = (g2/g1)**p_ref
actual_ratio = delta_tau_frac(g2) / delta_tau_frac(g1)
sublinear_ok = abs(actual_ratio - expected_ratio) / expected_ratio < 0.01
falsified = not sublinear_ok

# ===============================================================
# 7) VISUALIZATION
# ===============================================================
fig, ax = plt.subplots(figsize=(10, 6), dpi=150)
fig.patch.set_facecolor('#050814')
ax.set_facecolor('#050814')

ax.loglog(g_vals, np.abs(d_tau_frac), '#fbbf24', lw=2, label='|Δτ/τ|')
ax.loglog(g_vals, np.abs(d_sigma_xx_frac), '#60a5fa', lw=2, ls='--', label='|Δσ_xx/σ_xx|')
ax.loglog(g_vals, np.abs(d_sigma_xy_frac), '#a78bfa', lw=2, ls=':', label='|Δσ_xy/σ_xy|')

ax.axvline(g1, color='#94a3b8', ls='--', alpha=0.5)
ax.axvline(g2, color='#94a3b8', ls='--', alpha=0.5)
ax.text(g1*1.2, 1e-7, 'g_a=10⁻¹²', color='#94a3b8', fontsize=9)
ax.text(g2*1.2, 1e-7, 'g_a=10⁻¹⁰', color='#94a3b8', fontsize=9)

ax.set_xlabel('Axion coupling g_a', color='#94a3b8', fontsize=12)
ax.set_ylabel('Fractional change', color='#94a3b8', fontsize=12)
ax.set_title('STUR Graphene-Axion Response', color='#ffaa00', fontsize=14)
ax.tick_params(colors='#94a3b8')
ax.grid(True, which='both', ls=':', color='#1f2a44', alpha=0.5)
ax.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white', fontsize=10)

for spine in ax.spines.values():
    spine.set_color('#334155')

plt.tight_layout()
plt.savefig(OUT_PNG, facecolor='#050814', dpi=150, bbox_inches='tight')
plt.show()

# ===============================================================
# 8) JSON OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "graphene-axion",
        "version": "",
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "equation": "Δτ/τ = C(g_a/g₀)^p"
    },
    "graphene": {
        "T_K": T,
        "n_cm2": n_cm2,
        "E_F_J": float(E_F),
        "m_eff_kg": float(m_eff),
        "tau0_s": tau0,
        "B_T": B,
        "sigma_xx0_S_per_m": float(sigma_xx0),
        "sigma_xy0_S_per_m": float(sigma_xy0)
    },
    "axion_model": {
        "g0_ref": g0_ref,
        "C_ref": C_ref,
        "p_ref": p_ref
    },
    "test_points": {
        "g_a_1e-12": pt1,
        "g_a_1e-10": pt2
    },
    "falsification": {
        "criteria": [
            "Δσ_xy @ g₁ exceeds 10⁻⁴",
            "Linear scaling (p=1)",
            "No effect at g₂"
        ],
        "sublinear_check": {
            "expected_ratio": float(expected_ratio),
            "actual_ratio": float(actual_ratio),
            "pass": bool(sublinear_ok)
        },
        "triggered": ["Exclusion-2" if falsified else None],
        "falsified": falsified
    },
    "outputs": {"json": OUT_JSON, "png": OUT_PNG}
}

with open(OUT_JSON, "w") as f:
    json.dump(report, f, indent=2)

print("=== STUR Graphene-Axion ===")
print(f"At g_a=10⁻¹²: Δτ/τ = {pt1['delta_tau_over_tau']:.2e}")
print(f"At g_a=10⁻¹⁰: Δτ/τ = {pt2['delta_tau_over_tau']:.2e}")
print(f"Sublinear:    {'PASS' if sublinear_ok else 'FAIL'}")
print(f"FALSIFIED =   {falsified}")
print(f"Outputs: {OUT_PNG}, {OUT_JSON}")
