Executable Simulation

STUR-X 5D negative energy simulation with Monte Carlo averaging. Copy and run in any Python 3 environment.

#!/usr/bin/env python3
# ===============================================================
# STUR-X 5D Negative Energy (v9.2)
#
# Derives exotic stress-energy from S_X(r,chi) modulator
# Computes entropy bounds and X-coherence metric
#
# Outputs:
#   - stur_negative.json  (profiles and diagnostics)
#   - stur_negative.png   (force, energy, entropy plots)
#
# Derived from Master Action with 5D uplift
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, time

plt.style.use("dark_background")
np.random.seed(42)

OUT_JSON = "stur_negative.json"
OUT_PNG  = "stur_negative.png"

# ===============================================================
# 1) CONSTANTS (SI)
# ===============================================================
k_e = 8.9875e9
G = 6.6743e-11
q = 1.602e-19
m = 1.673e-27
alpha_s = 1e-26
sigma = 1e50
lambda_ = 1e-15
L_0 = 1e-35
hbar = 1.055e-34
c = 3e8
k_B = 1.38e-23
ell_P = 1.616e-35
m_P = 2.176e-8
rho_P = m_P * c**2 / ell_P**3

chi_0 = 1.0
chi_c = 2.0
chi_max = 4.0
N_chi = 21
num_runs = 8

r_values = np.logspace(-35, -10, num=15)
chi_values = np.linspace(-chi_max, chi_max, N_chi)

# ===============================================================
# 2) STUR-X MODULATORS
# ===============================================================
def S0(r):
    return np.tanh(r / L_0) * (1.0 - np.exp(-r / L_0))

def S_X(r, chi):
    s0 = S0(r)
    env = np.exp(-(chi / chi_0)**2)
    curv_pen = 1.0 / (1.0 + (chi / chi_c)**2)
    return s0 * env * curv_pen

def kappa_X(SX_val, rho_val, p_r_val):
    denom = 1.0 + abs(rho_val + p_r_val) / rho_P
    return float((SX_val * SX_val) / denom)

# ===============================================================
# 3) MONTE CARLO AVERAGING
# ===============================================================
b = 1e-15  # throat radius
V_0 = 1e-45

avg_S = []
avg_rho_pr = []
avg_kappa = []

for r in r_values:
    S_list = []
    rho_pr_list = []
    kappa_list = []

    for _ in range(num_runs):
        for chi in chi_values:
            sx = S_X(r, chi)
            # Simplified energy density and pressure
            rho_v = (b**2) / (8.0 * np.pi * (r**2 + b**2)**2)
            pr_v = -sx * alpha_s * sigma * r / (4.0 * np.pi * r**2 * V_0) if r > 0 else 0
            rho_pr = rho_v + pr_v

            S_list.append(sx)
            rho_pr_list.append(rho_pr)
            kappa_list.append(kappa_X(sx, rho_v, pr_v))

    avg_S.append(np.mean(S_list))
    avg_rho_pr.append(np.mean(rho_pr_list))
    avg_kappa.append(np.mean(kappa_list))

# ===============================================================
# 4) VISUALIZATION
# ===============================================================
fig, axes = plt.subplots(1, 3, figsize=(14, 4), dpi=150)
fig.patch.set_facecolor('#050814')

# S_X modulator
ax1 = axes[0]
ax1.set_facecolor('#050814')
ax1.loglog(r_values, avg_S, color='#4ade80', lw=2)
ax1.set_xlabel('r (m)', color='#94a3b8')
ax1.set_ylabel('S_X (chi-averaged)', color='#94a3b8')
ax1.set_title('STUR-X 5D Modulator', color='#ffaa00')
ax1.tick_params(colors='#94a3b8')
ax1.grid(alpha=0.3, color='#1f2a44')
for spine in ax1.spines.values():
    spine.set_color('#334155')

# rho + p_r
ax2 = axes[1]
ax2.set_facecolor('#050814')
ax2.semilogx(r_values, avg_rho_pr, color='#f472b6', lw=2)
ax2.axhline(0, ls='--', color='#94a3b8', alpha=0.5)
ax2.set_xlabel('r (m)', color='#94a3b8')
ax2.set_ylabel('rho + p_r (J/m^3)', color='#94a3b8')
ax2.set_title('Energy Condition', color='#ffaa00')
ax2.tick_params(colors='#94a3b8')
ax2.grid(alpha=0.3, color='#1f2a44')
for spine in ax2.spines.values():
    spine.set_color('#334155')

# X-coherence
ax3 = axes[2]
ax3.set_facecolor('#050814')
ax3.semilogx(r_values, avg_kappa, color='#fbbf24', lw=2)
ax3.set_xlabel('r (m)', color='#94a3b8')
ax3.set_ylabel('kappa_X', color='#94a3b8')
ax3.set_title('X-Coherence Metric', color='#ffaa00')
ax3.tick_params(colors='#94a3b8')
ax3.grid(alpha=0.3, color='#1f2a44')
for spine in ax3.spines.values():
    spine.set_color('#334155')

plt.tight_layout()
plt.savefig(OUT_PNG, facecolor='#050814', dpi=150, bbox_inches='tight')
plt.show()

# ===============================================================
# 5) JSON OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "negative-energy-5d",
        "version": "v9.2",
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "equation": "S_X(r,chi) = S0(r) * exp(-(chi/chi0)^2) / (1+(chi/chi_c)^2)"
    },
    "geometry": "M4(-+++) + chi",
    "params": {
        "chi_0": chi_0,
        "chi_c": chi_c,
        "N_chi": N_chi,
        "num_runs": num_runs
    },
    "profiles": {
        "r_m": [float(v) for v in r_values],
        "S_X_avg": [float(v) for v in avg_S],
        "rho_plus_pr": [float(v) for v in avg_rho_pr],
        "kappa_X": [float(v) for v in avg_kappa]
    },
    "outputs": {"json": OUT_JSON, "png": OUT_PNG}
}

with open(OUT_JSON, "w") as f:
    json.dump(report, f, indent=2)
