Executable Simulation

STUR universal immunity algorithm simulating flu, cancer, and HIV immune regimes. Copy and run in any Python 3 environment.

#!/usr/bin/env python3
# ===============================================================
# STUR Universal Immunity Algorithm
#
# Implements the STUR immune force field F_immune(r,t):
#   - Three regimes: flu (acute), cancer (suppressed), HIV (collapse)
#   - Curvature-regulated immune activation
#   - Chemotactic spin-torsion coupling
#
# Derived from Master Action torsion term
# ===============================================================

import numpy as np
import json
import matplotlib.pyplot as plt

# Scales and constants
L0 = 0.5e-6           # curvature scale (m)
lambda_sig = 1.0e-9   # signaling correlation length
ell_chem = 5.0e-6     # chemotactic decay length

# Cell densities (cells/m^3)
C_immune_base = 5e12
P_path_flu = 5e13
P_path_cancer = 5e11
P_path_HIV = 1e12

# Chemotactic parameters
S_immune_mag = 2e-9
G_path_mag = 5e4
gamma_chem = 5e-13
sigma_prolif = 1e-2
kappa_base = 5e-7

# STUR regulator
def S_reg(r, t, L0=L0, kappa=kappa_base):
    x = r / L0
    return np.tanh(x) * (1.0 - np.exp(-x)) * np.exp(-kappa * t)

# Immune force field
def immune_force_radial(r, t, params):
    C_immune = params["C_immune"]
    C_reg = params["C_reg"]
    P_path = params["P_path"]
    k_innate = params["k_innate"]
    k_suppress = params["k_suppress"]
    k_adaptive = params["k_adaptive"]
    kappa_scale = params["kappa_scale"]
    gamma_scale = params["gamma_chem_scale"]

    r_eff = np.maximum(r, 1e-9)
    S = S_reg(r_eff, t, L0=L0, kappa=kappa_base * kappa_scale)

    innate_term = k_innate * C_immune * P_path / (r_eff**2)
    suppress_term = k_suppress * C_reg * P_path / (r_eff**2)
    adaptive_term = k_adaptive * ((1.0 - np.exp(-r_eff / lambda_sig))/(r_eff**2) + sigma_prolif * r_eff)

    F_rad = S * (innate_term - suppress_term + adaptive_term)
    F_chem = gamma_chem * gamma_scale * np.exp(-r_eff/ell_chem) * S_immune_mag * G_path_mag

    return F_rad + F_chem

# Disease mode parameters
modes = {
    "flu": {
        "C_immune": C_immune_base,
        "C_reg": 0.4 * C_immune_base,
        "P_path": P_path_flu,
        "k_innate": 4e-14,
        "k_suppress": 2e-14,
        "k_adaptive": 5e-14,
        "kappa_scale": 1.0,
        "gamma_chem_scale": 1.0
    },
    "cancer": {
        "C_immune": 0.6 * C_immune_base,
        "C_reg": 0.9 * C_immune_base,
        "P_path": P_path_cancer,
        "k_innate": 3e-14,
        "k_suppress": 6e-14,
        "k_adaptive": 4e-14,
        "kappa_scale": 0.3,
        "gamma_chem_scale": 0.7
    },
    "HIV": {
        "C_immune": 0.3 * C_immune_base,
        "C_reg": 0.7 * C_immune_base,
        "P_path": P_path_HIV,
        "k_innate": 2e-14,
        "k_suppress": 5e-14,
        "k_adaptive": 2e-14,
        "kappa_scale": 0.5,
        "gamma_chem_scale": 0.5
    }
}

# Grids
r_vals = np.logspace(-8, -4.3, 128)
Tmax = 3600.0
t_vals = np.linspace(0.0, Tmax, 160)
R, T = np.meshgrid(r_vals, t_vals, indexing="ij")

# Compute force maps
force_maps = {}
for name, params in modes.items():
    F = immune_force_radial(R, T, params)
    force_maps[name] = F

# Compute metrics
summary = {}
for name, F in force_maps.items():
    F_int_r = np.trapz(np.abs(F), t_vals, axis=1)
    clearance_index = float(np.trapz(F_int_r, r_vals))
    idx_r_peak = int(np.argmax(F_int_r))
    r_peak = float(r_vals[idx_r_peak])

    summary[name] = {
        "clearance_index": clearance_index,
        "r_peak_m": r_peak
    }

print("=== STUR Universal Immunity ===")
for name, metrics in summary.items():
    print(f"{name}: clearance={metrics['clearance_index']:.2e}, r_peak={metrics['r_peak_m']*1e6:.2f} um")

# Plots
fig, axes = plt.subplots(1, 2, figsize=(14, 5))
fig.patch.set_facecolor('#050814')

for ax in axes:
    ax.set_facecolor('#050814')
    ax.tick_params(colors='#94a3b8')
    ax.grid(True, alpha=0.3, color='#1f2a44')
    for spine in ax.spines.values():
        spine.set_color('#334155')

# Force at r=10um vs time
r_probe = 10e-6
idx_probe = int(np.argmin(np.abs(r_vals - r_probe)))
colors = {'flu': '#4ade80', 'cancer': '#fbbf24', 'HIV': '#f472b6'}

for name in ["flu", "cancer", "HIV"]:
    F_probe = force_maps[name][idx_probe, :]
    axes[0].plot(t_vals/60.0, F_probe, label=name, color=colors[name], lw=2)

axes[0].axhline(0, color='gray', ls='--', alpha=0.5)
axes[0].set_xlabel("Time (min)", color='#94a3b8')
axes[0].set_ylabel("F_immune (act/m^2)", color='#94a3b8')
axes[0].set_title("Immune Force at r=10um", color='#ffaa00')
axes[0].legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')

# Time-integrated profile
for name in ["flu", "cancer", "HIV"]:
    F = force_maps[name]
    F_int_r = np.trapz(np.abs(F), t_vals, axis=1)
    axes[1].loglog(r_vals*1e6, F_int_r, label=name, color=colors[name], lw=2)

axes[1].set_xlabel("Radius r (um)", color='#94a3b8')
axes[1].set_ylabel("Integrated |F| (act/m)", color='#94a3b8')
axes[1].set_title("Time-Integrated Immune Response", color='#ffaa00')
axes[1].legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')

fig.suptitle('STUR Universal Immunity Algorithm', color='#ffaa00', fontsize=14)
plt.tight_layout()
plt.show()
