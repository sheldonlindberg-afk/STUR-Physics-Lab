Golden ratio attractor simulation with STUR feedback coupling. Copy and run in any Python 3 environment.

#!/usr/bin/env python3
# ===============================================================
# STUR Unity Attractor Simulation — Complete EFT Framework
#
# Demonstrates R* = E/(I*Phi) -> 1 from XCRM coupling
# ONE input: M_Planck (sets all scales)
# geometric framework (simulation uses illustrative values)
#
# Outputs:
#   - stur_goldenratio.json  (attractor statistics)
#   - stur_goldenratio.png   (time evolution plot)
#
# Note: The "golden ratio" naming is historical. The actual
# attractor value is unity (1), not phi (1.618...).
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, time

plt.style.use("dark_background")
np.random.seed(42)

OUT_JSON = "stur_goldenratio.json"
OUT_PNG  = "stur_goldenratio.png"

# ===============================================================
# 1) PARAMETERS
# ===============================================================
E0   = 1.0          # normalized energy unit
I0   = 1.0          # normalized information flow
Phi0 = 1.0          # normalized geometric flux

# Oscillation & coupling constants
omega_E   = 2.0 * np.pi * 0.25   # Hz equivalent
omega_I   = 2.0 * np.pi * 0.27
omega_Phi = 2.0 * np.pi * 0.23
k_couple  = 0.15                 # coupling factor for feedback
noise     = 0.02

# Time domain
t = np.linspace(0, 60, 2000)  # 60 s, fine resolution

# ===============================================================
# 2) DYNAMICS
# ===============================================================
# Coherent oscillations with slight phase drifts
E   = E0   * (1 + 0.3*np.sin(omega_E*t))
I   = I0   * (1 + 0.3*np.sin(omega_I*t + np.pi/3))
Phi = Phi0 * (1 + 0.3*np.sin(omega_Phi*t + np.pi/5))

# Feedback coupling: small self-correction toward unity
for _ in range(3):
    R = E / (I*Phi)
    E -= k_couple * (R - 1) * E + noise*np.random.randn(len(E))*1e-2

# Recompute final ratio
R_star = E / (I * Phi)

# ===============================================================
# 3) STATISTICS
# ===============================================================
R_mean = np.mean(R_star)
R_std  = np.std(R_star)

# ===============================================================
# 4) VISUALIZATION
# ===============================================================
fig, ax = plt.subplots(figsize=(10, 5), dpi=150)
fig.patch.set_facecolor('#050814')
ax.set_facecolor('#050814')

ax.plot(t, R_star, color="#ffaa00", lw=1.5, label=r"$R_\ast = E/(I\Phi)$")
ax.axhline(1, color="#22d3ee", ls="--", lw=1.5, label="Perfect Coherence (R*=1)")
ax.fill_between(t, 1-R_std, 1+R_std, color="#22d3ee", alpha=0.1)

ax.set_xlabel("Time (s)", color='#94a3b8')
ax.set_ylabel("R* ratio", color='#94a3b8')
ax.set_title("STUR Unity Attractor (Complete EFT)", color='#ffaa00', fontsize=14)
ax.tick_params(colors='#94a3b8')
ax.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')
ax.grid(alpha=0.3, color='#1f2a44')
for spine in ax.spines.values():
    spine.set_color('#334155')

plt.tight_layout()
plt.savefig(OUT_PNG, facecolor='#050814', dpi=150, bbox_inches='tight')
plt.show()

# ===============================================================
# 5) JSON OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "unity-attractor",
        "version": "4.0",
        "framework": "ONE input (M_Planck), Z₃ geometric framework",

        "equation": "R* = E / (I * Phi) -> 1"
    },
    "params": {
        "E0": float(E0),
        "I0": float(I0),
        "Phi0": float(Phi0),
        "k_couple": float(k_couple),
        "noise": float(noise),
        "omega_E": float(omega_E),
        "omega_I": float(omega_I),
        "omega_Phi": float(omega_Phi)
    },
    "results": {
        "R_mean": float(R_mean),
        "R_std": float(R_std),
        "E_mean": float(np.mean(E)),
        "I_mean": float(np.mean(I)),
        "Phi_mean": float(np.mean(Phi))
    },
    "interpretation": "R* -> 1 (unity, not golden ratio) indicates energy-information-geometry equilibrium",
    "outputs": {"json": OUT_JSON, "png": OUT_PNG}
}

with open(OUT_JSON, "w") as f:
    json.dump(report, f, indent=2)

print("=== STUR Unity Attractor (Complete EFT) ===")
print(f"R* mean:  {R_mean:.4f}  (target: 1.0, not phi)")

print(f"Framework: ONE input (M_Planck), Z₃ geometric framework")
