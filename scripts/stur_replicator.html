Stress-failure envelope visualization with STUR replicator dynamics. Copy and run in any Python 3 environment.

#!/usr/bin/env python3
# ===============================================================
# STUR Replicator - Stress-Failure Envelope (v9.2)
#
# Derives failure-mode boundaries from orbifold-derived V(R) (§2.5)
# Classifications: stable, power-limited, biology-limited
#
# Outputs:
#   - stur_replicator.json  (classification results)
#   - stur_replicator.png   (failure envelope plots)
#
# Derived from orbifold-derived V(R) potential (Foundations §2.5)
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, time

# ===============================================================
# 1) SAMPLE DATA (from Step 1 replicator chain)
# ===============================================================
results = [
    {"class":"inert","mass_kg":1e-6,"t_s":60,"failure_mode":"power","ratio_vs_mc2":1e-12,"dS_worst":0.1},
    {"class":"inert","mass_kg":1e-3,"t_s":60,"failure_mode":"stable","ratio_vs_mc2":1e-9,"dS_worst":0.05},
    {"class":"food","mass_kg":0.01,"t_s":60,"failure_mode":"biology","ratio_vs_mc2":5e-9,"dS_worst":0.2},
    {"class":"organ","mass_kg":1.0,"t_s":60,"failure_mode":"power","ratio_vs_mc2":1e-6,"dS_worst":0.3},
    {"class":"organ","mass_kg":5.0,"t_s":60,"failure_mode":"stable","ratio_vs_mc2":5e-7,"dS_worst":0.15}
]

OUT_JSON = "stur_replicator.json"
OUT_PNG  = "stur_replicator.png"

# ===============================================================
# 2) HELPER FUNCTIONS
# ===============================================================
def get_arrays_for_time(results, t_sel):
    out = {}
    for r in results:
        if abs(r["t_s"] - t_sel) < 1e-9:
            c = r["class"]
            out.setdefault(c, {"mass":[], "ratio":[], "mode":[], "dS_worst":[]})
            out[c]["mass"].append(r["mass_kg"])
            out[c]["ratio"].append(r["ratio_vs_mc2"])
            out[c]["mode"].append(r["failure_mode"])
            out[c]["dS_worst"].append(r["dS_worst"])
    for c in out:
        idx = np.argsort(out[c]["mass"])
        for k in out[c]:
            out[c][k] = list(np.array(out[c][k])[idx])
    return out

mode_color = {"stable": "#4ade80", "power": "#fbbf24", "biology": "#f472b6"}
mode_marker = {"stable": "o", "power": "^", "biology": "s"}
mode_numeric = {"power": 0, "biology": 1, "stable": 2}

# ===============================================================
# 3) VISUALIZATION
# ===============================================================
fig, axes = plt.subplots(1, 3, figsize=(14, 4), dpi=150)
fig.patch.set_facecolor('#050814')

t = 60
d = get_arrays_for_time(results, t)

# Plot 1: Failure-Mode Envelope
ax1 = axes[0]
ax1.set_facecolor('#050814')
for i, c in enumerate(["inert", "food", "organ"]):
    if c not in d:
        continue
    m = np.array(d[c]["mass"])
    y = np.array([mode_numeric[x] for x in d[c]["mode"]])
    o = np.argsort(m)
    ax1.step(m[o], y[o] + 0.05*i, where="post", label=f"{c}(t={t}s)", lw=2)
ax1.set_xscale("log")
ax1.set_yticks([0, 1, 2])
ax1.set_yticklabels(["power", "bio", "stable"])
ax1.set_xlabel("Mass (kg)", color='#94a3b8')
ax1.set_ylabel("Mode", color='#94a3b8')
ax1.set_title(f"Failure-Mode Envelope @ t={t}s", color='#ffaa00')
ax1.tick_params(colors='#94a3b8')
ax1.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white', fontsize=8)
ax1.grid(alpha=0.3, color='#1f2a44')
for spine in ax1.spines.values():
    spine.set_color('#334155')

# Plot 2: Energy Fraction
ax2 = axes[1]
ax2.set_facecolor('#050814')
for c in ["inert", "food", "organ"]:
    if c not in d:
        continue
    for m, r, fm in zip(d[c]["mass"], d[c]["ratio"], d[c]["mode"]):
        ax2.scatter(m, r, c=mode_color[fm], marker=mode_marker[fm],
                    edgecolors="black", alpha=0.8, s=80)
ax2.set_xscale("log")
ax2.set_yscale("log")
ax2.set_xlabel("Mass (kg)", color='#94a3b8')
ax2.set_ylabel("E_need / (m c^2)", color='#94a3b8')
ax2.set_title(f"Energy Fraction @ t={t}s", color='#ffaa00')
ax2.tick_params(colors='#94a3b8')
ax2.grid(alpha=0.3, color='#1f2a44')
for spine in ax2.spines.values():
    spine.set_color('#334155')

# Plot 3: Entropy Gradient
ax3 = axes[2]
ax3.set_facecolor('#050814')
for c in ["inert", "food", "organ"]:
    if c not in d:
        continue
    for m, ds, fm in zip(d[c]["mass"], d[c]["dS_worst"], d[c]["mode"]):
        ax3.scatter(m, ds, c=mode_color[fm], marker=mode_marker[fm],
                    edgecolors="black", alpha=0.8, s=80)
ax3.set_xscale("log")
ax3.set_xlabel("Mass (kg)", color='#94a3b8')
ax3.set_ylabel("Worst-case dS_eff", color='#94a3b8')
ax3.set_title(f"Entropy Gradient @ t={t}s", color='#ffaa00')
ax3.tick_params(colors='#94a3b8')
ax3.grid(alpha=0.3, color='#1f2a44')
for spine in ax3.spines.values():
    spine.set_color('#334155')

plt.tight_layout()
plt.savefig(OUT_PNG, facecolor='#050814', dpi=150, bbox_inches='tight')
plt.show()

# ===============================================================
# 4) SUMMARY STATISTICS
# ===============================================================
def summarize_results(results):
    out = {"n_total": len(results), "classes": {},
           "times_s": sorted({float(r["t_s"]) for r in results})}
    for c in sorted({r["class"] for r in results}):
        s = [r for r in results if r["class"] == c]
        out["classes"][c] = {"count": len(s), "failure_modes": {}}
        for fm in ["stable", "power", "biology"]:
            s2 = [r for r in s if r["failure_mode"] == fm]
            if not s2:
                continue
            m = np.array([r["mass_kg"] for r in s2])
            R = np.array([r["ratio_vs_mc2"] for r in s2])
            D = np.array([r["dS_worst"] for r in s2])
            out["classes"][c]["failure_modes"][fm] = {
                "count": len(s2),
                "mass_kg_min": float(m.min()),
                "mass_kg_max": float(m.max()),
                "ratio_vs_mc2_median": float(np.median(R)),
                "dS_worst_mean": float(D.mean())
            }
    return out

# ===============================================================
# 5) JSON OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "stress-failure-envelope",
        "version": "v9.2",
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "equation": "E_need = integral V(R) dV + dS_eff * T"
    },
    "summary": summarize_results(results),
    "outputs": {"json": OUT_JSON, "png": OUT_PNG}
}

with open(OUT_JSON, "w") as f:
    json.dump(report, f, indent=2)

print("=== STUR Replicator Stress-Failure [v9.2] ===")
print(f"Total samples: {report['summary']['n_total']}")
print(f"Classes: {list(report['summary']['classes'].keys())}")
print(f"Outputs: {OUT_PNG}, {OUT_JSON}")
