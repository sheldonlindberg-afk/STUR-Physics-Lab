#!/usr/bin/env python3
# ===============================================================
# STUR TOE Proof — Macro-Quantum Isochrony (v9.2)
#
# Tests unified curvature: T(m) = 2π√(m/k_target)
# Same k produces T(1kg)≈15.7h and T(proton)≈2.31ns
#
# Outputs:
#   - stur_toe.json  (full results + falsification flag)
#   - stur_toe.png   (period vs mass + isochrony plot)
#
# Falsified if: T ∝ √m fails or A back-solve is inconsistent
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, time

# ===============================================================
# 1) STUR RESISTANCE FIELD
# ===============================================================
def S(r, L0):
    """STUR resistance modulator S(r) from V(R)"""
    x = r / L0
    return np.tanh(x) * (1.0 - np.exp(-x))

def dS_dr(r, L0):
    """Derivative dS/dr"""
    x = r / L0
    sech2 = 1.0 / np.cosh(x)**2
    return (sech2 * (1.0 - np.exp(-x)) + np.tanh(x) * np.exp(-x)) / L0

# ===============================================================
# 2) CURVATURE LAW
# ===============================================================
def stur_curvature(A, r0, L0):
    """Effective curvature k(r0) from geometry + interaction scale A"""
    Sr  = S(r0, L0)
    dSr = dS_dr(r0, L0)
    return (2.0 * A * Sr) / (r0**3) - (A * dSr) / (r0**2)

k_target = 1.234e-8  # N/m (target curvature)

OUT_JSON = "stur_toe.json"
OUT_PNG  = "stur_toe.png"

# ===============================================================
# 3) PERIOD FUNCTION
# ===============================================================
def period(m, k):
    """Oscillator period T(m) = 2π√(m/k)"""
    return 2.0 * np.pi * np.sqrt(m / k)

# Test masses
m_macro = 1.0              # kg
m_p = 1.6726e-27           # proton mass (kg)

T_macro = period(m_macro, k_target)
T_p = period(m_p, k_target)
f_p = 1.0 / T_p

# ===============================================================
# 4) BACK-SOLVE FOR INTERACTION SCALE A
# ===============================================================
r0 = 0.10  # equilibrium radius (m)
L0 = 0.10  # STUR length scale (m)

Sr0 = S(r0, L0)
dSr0 = dS_dr(r0, L0)
geom_factor = (2.0 * Sr0) / (r0**3) - dSr0 / (r0**2)

A_needed = k_target / geom_factor
k_check = stur_curvature(A_needed, r0, L0)

# Curvature profile for isochrony plateau visualization
r_grid = np.linspace(0.02, 0.25, 400)
k_grid = stur_curvature(A_needed, r_grid, L0)

# ===============================================================
# 5) MASS SCAN
# ===============================================================
m_grid = np.logspace(-30, 5, 400)
T_grid = period(m_grid, k_target)
isochrony_ratio = T_grid / np.sqrt(m_grid)

# Check isochrony constancy
iso_mean = np.mean(isochrony_ratio)
iso_std = np.std(isochrony_ratio)
iso_consistent = (iso_std / iso_mean) < 1e-10  # should be constant

# ===============================================================
# 6) FALSIFICATION CHECK
# ===============================================================
k_consistent = abs(k_check - k_target) / k_target < 1e-6
falsified = not (k_consistent and iso_consistent)

# ===============================================================
# 7) VISUALIZATION
# ===============================================================

fig.patch.set_facecolor('#050814')

# Panel 1: Period vs Mass
ax1 = fig.add_subplot(1, 3, 1)
ax1.set_facecolor('#050814')
ax1.loglog(m_grid, T_grid, '#60a5fa', lw=2)
ax1.scatter([m_macro], [T_macro], s=60, c='#4ade80', zorder=5, label='1 kg')
ax1.scatter([m_p], [T_p], s=60, c='#f472b6', zorder=5, label='proton')
ax1.set_xlabel('mass m [kg]', color='#94a3b8')
ax1.set_ylabel('period T [s]', color='#94a3b8')
ax1.set_title('T(m) with fixed k', color='#ffaa00', fontsize=12)
ax1.tick_params(colors='#94a3b8')
ax1.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white', fontsize=8)
ax1.grid(alpha=0.25, which='both', color='#1f2a44')
for spine in ax1.spines.values():
    spine.set_color('#334155')

# Panel 2: Curvature profile
ax2 = fig.add_subplot(1, 3, 2)
ax2.set_facecolor('#050814')
ax2.plot(r_grid, k_grid, '#fbbf24', lw=2)
ax2.axvline(r0, ls='--', lw=1, color='#94a3b8', alpha=0.7)
ax2.set_xlabel('radius r [m]', color='#94a3b8')
ax2.set_ylabel('k(r) [N/m]', color='#94a3b8')
