Executable Simulation

Python simulation evolving the arc for 5 seconds with multi-force coupling.

#!/usr/bin/env python3
"""
STUR Resistance Arc Simulator
====================================
Tests the resistance modulator S(r) from the Master Action potential term.

Derives from: S[\mathbf{R},g,X] → V(R) → radial modulation S(r)

Outputs:
  - Effective stiffness (static & feedback)
  - Trap frequency and oscillation period
  - Binding energy and coherence ratio
  - JSON results with falsification flag
"""

import numpy as np
import matplotlib.pyplot as plt
import json, time

# ===============================================================
# FUNDAMENTAL + STUR PARAMETERS
# ===============================================================
mu0 = 4*np.pi*1e-7
mB  = 1e-6          # effective "arc mass" [kg]
q   = 1.6e-19       # unit charge [C]

ke  = 8.9875e9      # Coulomb constant
G   = 6.6743e-11    # Newton G
as0 = 1e-24         # strong-like coupling scale
sig = 1e3           # string-like restoring coefficient
lam = 1e-15         # strong-force range [m]
L0  = 1e-10         # STUR modulation length [m] = √2 L_X / σ_R

B0, b1, b2 = 5, 500, 5e4
Abl = 3.1e-4        # effective area for magnetic pressure [m²]
Vb  = 4.19e-6       # confining volume scale [m³]

U0 = 3e5            # external potential scale
Ls = 1e-2           # external decay length [m]

kB = 1.380649e-23
T0 = 300.0          # ambient temperature [K]

c0  = 6e-3          # base damping coefficient
eps = 1e-15         # gap floor [m]
d0  = 1e-14         # initial gap [m]
kfb = 1e9           # feedback stiffness gain

kmin, kmax = 1e3, 1e4   # stiffness clipping

# ===============================================================
# STUR RADIAL MODULATOR [from V(R) term]
# S(r) = tanh(r/L₀) · (1 - exp(-r/L₀))
# ===============================================================
def S(r):
    """STUR radial resistance modulator."""
    return np.tanh(r/L0) * (1.0 - np.exp(-r/L0))

def Fstur(r):
    """
    Net STUR force: EM + gravity + strong-like + string
    All gated by S(r) from the potential term.
    """
    return S(r)*(
        ke*q*q/r**2           # Coulomb
        - G*mB*mB/r**2        # gravity
        - as0*(1.0 - np.exp(-r/lam))/r**2  # strong-like
        + sig*r               # string restoration
    )

# ===============================================================
# MAGNETIC PRESSURE & EXTERNAL CONFINEMENT
# ===============================================================
def dPB():
    return (B0*b1)/mu0

def d2PB():
    return (b1**2 + 2*b2*B0)/mu0

Fin0 = -Abl*dPB()
Kin  = -Abl*d2PB()

def Fout(r):
    """External confinement force from decaying potential."""
    return -Vb*U0*np.exp(-r/Ls)*(2.0/r**3 + 1.0/(Ls*r**2))

def Kout(r):
    """Curvature of external confinement."""
    x = 1.0/Ls
    return -Vb*U0*np.exp(-r/Ls)*(-6.0/r**4 - 4.0*x/r**3 - x**2/r**2)

# ===============================================================
# EFFECTIVE STIFFNESS
# ===============================================================
r0 = 1e-2
Keff_stat = -(Kin + Kout(r0))
kstat = np.clip(Keff_stat, kmin, kmax)

def keff(d):
    """Stiffness with feedback: k_static + k_fb/(d + ε)"""
    return float(np.clip(kstat + kfb/(d + eps), kmin, kmax))

def damp(d):
    """Distance-dependent damping."""
    return c0 * (d/d0)**0.5

# ===============================================================
# TIME EVOLUTION (5 seconds)
# ===============================================================
dt = 1e-7        # 0.1 µs timestep
tf = 5.0         # 5 seconds
N  = int(tf/dt)

# Downsample for storage
Nsample = 500_000
skip    = max(1, N//Nsample)
Nt      = Nsample

x = np.zeros(Nt)
v = np.zeros(Nt)
d_arr = np.zeros(Nt)
d_arr[0] = d0

xi, vi, di = 0.0, 0.0, d0

for i in range(N):
    k = keff(di)
    c = damp(di)
    a = (-k*xi - c*vi)/mB

    vi += a*dt
    xi += vi*dt
    di -= (di - eps)*1e7*dt

    if i % skip == 0:
        j = i//skip
        if j < Nt:
            x[j], v[j], d_arr[j] = xi, vi, di

# ===============================================================
# DERIVED QUANTITIES
# ===============================================================
xrms = float(np.sqrt(np.mean(x**2)))
dfin = float(d_arr[-1])
ftrap = float(np.sqrt(keff(dfin)/mB)/(2.0*np.pi))

rg = np.linspace(1e-12, 1e-9, 400)
Ebind = float(np.trapz(Fstur(rg), rg))
Ebind_eV = Ebind/1.602e-19

coh = float(np.exp(-(kB*T0)/abs(Ebind))) if Ebind != 0 else 0.0

# ===============================================================
# FALSIFICATION CHECK
# ===============================================================
# Exclusion-1: trap frequency within expected range
ftrap_expected = np.sqrt(kstat/mB)/(2*np.pi)
excl1 = abs(ftrap - ftrap_expected)/ftrap_expected > 0.1

# Exclusion-2: coherence should be positive and bounded
excl2 = coh < 0 or coh > 1

# Exclusion-3: feedback should increase stiffness
excl3 = keff(dfin) <= kstat

falsified = excl1 or excl2 or excl3

# ===============================================================
# OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "resistance-arc",
        "version": "",
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "equation": "S(r) = tanh(r/L₀)·(1 - exp(-r/L₀))"
    },
    "params": {
        "L0_m": L0,
        "mB_kg": mB,
        "kstat_N_per_m": float(kstat),
        "kfb": kfb
    },
    "results": {
        "k_static": float(kstat),
        "k_eff_final": keff(dfin),
        "f_trap_Hz": ftrap,
        "x_rms_m": xrms,
        "d_final_m": dfin,
        "E_bind_eV": Ebind_eV,
        "coherence_ratio": coh
    },
    "falsification": {
        "criteria": [
            "f_trap deviation > 10%",
            "Coherence ratio invalid",
            "No feedback stiffening"
        ],
        "triggered": [
            "Exclusion-1" if excl1 else None,
            "Exclusion-2" if excl2 else None,
            "Exclusion-3" if excl3 else None
        ],
        "falsified": falsified
    }
}

print("=== STUR Resistance Arc ===")
print(f"STUR Modulator: S(r) = tanh(r/L₀)·(1 - exp(-r/L₀))")
print(f"L₀ = {L0:.2e} m (= √2 L_X / σ_R)")
print()
print(f"Static stiffness:  {kstat:.2e} N/m")
print(f"Final stiffness:   {keff(dfin):.2e} N/m")
print(f"Trap frequency:    {ftrap:.2e} Hz")
print(f"Period:            {1e3/ftrap:.3f} ms")
print(f"RMS displacement:  {xrms*1e9:.3f} nm")
print(f"Binding energy:    {Ebind_eV:.2e} eV")
print(f"Coherence ratio:   {coh:.4f}")
print()
print(f"FALSIFIED: {falsified}")

# Plot
t_arr = np.linspace(0, tf, Nt)
plt.figure(figsize=(10, 4))

plt.subplot(1, 2, 1)
plt.plot(rg*1e9, Fstur(rg), color='#4ade80')
plt.xlabel("Radius (nm)")
plt.ylabel("Force (N)")
plt.title("STUR Resistance Force F(r)")
plt.grid(alpha=0.3)

plt.subplot(1, 2, 2)
plt.plot(t_arr, x*1e9, color='#60a5fa', lw=0.5)
plt.xlabel("Time (s)")
plt.ylabel("Displacement (nm)")
plt.title("Arc Oscillation (5 sec)")
plt.grid(alpha=0.3)

plt.tight_layout()
plt.savefig("stur_arc.png", dpi=150)
plt.show()

with open("stur_arc.json", "w") as f:
    json.dump(report, f, indent=2)
print(f"\nOutputs: stur_arc.png, stur_arc.json")
