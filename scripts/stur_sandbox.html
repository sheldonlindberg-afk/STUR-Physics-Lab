3 Numerical Methods

Translating PDEs to code requires careful discretization. We implement Crank-Nicolson — a semi-implicit method that's unconditionally stable.

Why Not Explicit Euler?

Simple forward Euler has a strict stability limit (CFL condition):

\[\Delta t < \frac{(\Delta x)^2}{2D} \quad \text{(explicit stability limit)}\]

This forces tiny timesteps for fine grids. Crank-Nicolson removes this restriction.

Crank-Nicolson: Semi-Implicit Averaging

Average the spatial operator between time levels \(n\) and \(n+1\):

\[\frac{R_i^{n+1} - R_i^n}{\Delta t} = \frac{1}{2}\Big[\mathcal{L}(R^n) + \mathcal{L}(R^{n+1})\Big]\]
Unconditionally stable for diffusion

This yields a tridiagonal system \(\mathbf{A}\cdot R^{n+1} = \mathbf{b}\), solved efficiently via Thomas algorithm.

Code Pattern: Thomas Algorithm (TDMA)

// Solve: a[i]·x[i-1] + b[i]·x[i] + c[i]·x[i+1] = d[i]
function solveTridiagonal(a, b, c, d) {
  const n = d.length;
  const cp = new Float64Array(n);  // c-prime
  const dp = new Float64Array(n);  // d-prime
  const x  = new Float64Array(n);  // solution

// Forward sweep
cp[0] = c[0] / b[0];
dp[0] = d[0] / b[0];
for (let i = 1; i < n; i++) {
const m = 1 / (b[i] - a[i] * cp[i-1]);
cp[i] = c[i] * m;
dp[i] = (d[i] - a[i] * dp[i-1]) * m;
}

// Back substitution
x[n-1] = dp[n-1];
for (let i = n-2; i >= 0; i–) {
x[i] = dp[i] - cp[i] * x[i+1];
}
return x;
}

def chi_squared(V_pred, V_obs, V_err):
    """Compute χ² for model fit"""
    return sum(((p - o) / e)**2 for p, o, e in zip(V_pred, V_obs, V_err))

def aic(chi2, k, n):
    """Akaike Information Criterion: AIC = χ² + 2k"""
    return chi2 + 2 * k

def bic(chi2, k, n):
    """Bayesian Information Criterion: BIC = χ² + k*ln(n)"""
    return chi2 + k * np.log(n)

def model_comparison(dL_vals, V_obs, V_err, models):
    """
    Compare multiple models using χ², AIC, BIC.
    
    models: dict of {'name': (V_pred_array, n_params)}
    Returns: dict with stats for each model, plus relative evidence
    
    Interpretation:
      - ΔAIC < 2: models indistinguishable
      - ΔAIC 4-7: some evidence for lower AIC
      - ΔAIC > 10: strong evidence for lower AIC
      - Same scale for BIC (often more conservative)
    """
    n = len(dL_vals)
    results = {}
    
    for name, (V_pred, k) in models.items():
        chi2 = chi_squared(V_pred, V_obs, V_err)
        results[name] = {
            'chi2': chi2,
            'chi2_per_dof': chi2 / (n - k),
            'aic': aic(chi2, k, n),
            'bic': bic(chi2, k, n),
            'n_params': k
        }
    
    # Compute relative evidence (Δ from best model)
    best_aic = min(r['aic'] for r in results.values())
    best_bic = min(r['bic'] for r in results.values())
    
    for name in results:
        results[name]['delta_aic'] = results[name]['aic'] - best_aic
        results[name]['delta_bic'] = results[name]['bic'] - best_bic
        # Akaike weight (relative likelihood)
        results[name]['akaike_weight'] = np.exp(-results[name]['delta_aic'] / 2)
    
    # Normalize weights
    total_weight = sum(r['akaike_weight'] for r in results.values())
    for name in results:
        results[name]['akaike_weight'] /= total_weight
    
    return results

def solve_tridiagonal(a, b, c, d):
    """
    Thomas algorithm for tridiagonal systems.
    Solves: a[i]·x[i-1] + b[i]·x[i] + c[i]·x[i+1] = d[i]
    """
    n = len(d)
    cp = np.zeros(n)
    dp = np.zeros(n)
    x = np.zeros(n)
    
    cp[0] = c[0] / b[0]
    dp[0] = d[0] / b[0]
    
    for i in range(1, n):
        denom = b[i] - a[i] * cp[i-1]
        m = 0.0 if abs(denom) < 1e-12 else 1.0 / denom
        cp[i] = c[i] * m
        dp[i] = (d[i] - a[i] * dp[i-1]) * m
    
    x[n-1] = dp[n-1]
    for i in range(n-2, -1, -1):
        x[i] = dp[i] - cp[i] * x[i+1]
    
    return x

class STURSolver:
    """
    1D STUR PDE solver with R ↔ T back-reaction.
    
    PDE: ∂R/∂t = D∇²R - (R-R_bg)/L_X + 2αTR
    Torsion: T ∝ κ(∂R/∂x)²
    """
    
    def __init__(self, N=100, L=10.0, D=0.5, L_X=2.0, 
                 alpha=0.1, kappa=1.0, R_bg=0.0, dt=0.5):
        self.N = N
        self.L = L
        self.dx = L / N
        self.D = D
        self.L_X = L_X
        self.alpha = alpha
        self.kappa = kappa
        self.R_bg = R_bg
        self.dt = dt
        
        self.R = np.zeros(N)
        self.T = np.zeros(N)
        self.timestep = 0
        self.initial_mass = 0.0
    
    def initialize_gaussian(self, R0=1.0, width_frac=0.1):
        """Initialize with Gaussian perturbation"""
        center = self.N // 2
        sigma = self.N * width_frac
        x = np.arange(self.N)
        self.R = self.R_bg + R0 * np.exp(-(x - center)**2 / (2*sigma**2))
        self.initial_mass = np.sum(self.R) * self.dx
        self.timestep = 0
    
    def compute_torsion(self):
        """Compute torsion from resistance gradients"""
        dRdx = np.zeros(self.N)
        dRdx[1:-1] = (self.R[2:] - self.R[:-2]) / (2*self.dx)
        dRdx[0] = dRdx[1]
        dRdx[-1] = dRdx[-2]
        self.T = self.kappa * dRdx**2
    
    def step(self):
        """Crank-Nicolson time step"""
        N, dx, D, L_X, alpha, dt = (
            self.N, self.dx, self.D, self.L_X, self.alpha, self.dt
        )
        sigma = D * dt / (2 * dx**2)
        
        a = np.zeros(N)
        b = np.zeros(N)
        c = np.zeros(N)
        d = np.zeros(N)
        
        self.compute_torsion()
        
        for i in range(1, N-1):
            laplacian = (self.R[i+1] - 2*self.R[i] + self.R[i-1]) / dx**2
            relaxation = -(self.R[i] - self.R_bg) / L_X
            source = 2 * alpha * self.T[i] * self.R[i]
            explicit = self.R[i] + (dt/2) * (D*laplacian + relaxation + source)
            
            a[i] = -sigma
            b[i] = 1 + 2*sigma + dt/(2*L_X)
            c[i] = -sigma
            d[i] = explicit
        
        # Dirichlet BCs
        b[0] = 1.0; d[0] = self.R_bg
        b[-1] = 1.0; d[-1] = self.R_bg
        
        self.R = solve_tridiagonal(a, b, c, d)
        self.timestep += 1
    
    def run(self, steps=100):
        for _ in range(steps):
            self.step()
    
    def verify(self):
        """Run verification diagnostics"""
        results = {}
        results['stable'] = np.all(np.isfinite(self.R))
        results['positive'] = np.all(self.R >= -1e-10)
        results['bounded'] = np.all(np.abs(self.R) < 100)
        
        current_mass = np.sum(self.R) * self.dx
        mass_change = abs(current_mass - self.initial_mass) / max(abs(self.initial_mass), 1e-10)
        # Tightened from 0.5 to 0.1 (10%) — Crank-Nicolson should conserve better
        results['mass_conserved'] = mass_change < 0.1
        results['mass_change_frac'] = mass_change
        
        return results
    
    def G_eff_ratio(self):
        """Compute G_eff/G at max R"""
        R_max = np.max(np.abs(self.R))
        return 1.0 / (1 + 16*np.pi * self.alpha * R_max**2)
    
    def to_dict(self):
        return {
            'params': {'N': self.N, 'L': self.L, 'D': self.D, 
                       'L_X': self.L_X, 'alpha': self.alpha, 'dt': self.dt},
            'state': {'timestep': self.timestep,
                      'R': self.R.tolist(), 'T': self.T.tolist()},
            'diagnostics': self.verify(),
            'G_eff_ratio': self.G_eff_ratio()
        }
