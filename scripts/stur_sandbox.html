Python – STUR 5D Sandbox Script (Copy & Run)

  

    This script reproduces a simple 5D worldline in pure Python:
    NumPy + Matplotlib + JSON only (JupyterLite safe).
  


  
#!/usr/bin/env python3
# ============================================================
# STUR 5D SANDBOX – v1.0
# 3 space + t + χ torsion-time
# - Real constants (cosmology + EM)
# - Simple 5D worldline
# - JupyterLite-safe (NumPy + Matplotlib + JSON)
# ============================================================

import numpy as np
import matplotlib.pyplot as plt
import json

plt.style.use("dark_background")

# ------------------------------------------------------------
# 1. Physical constants (approximate, real-world)
# ------------------------------------------------------------
c = 2.99792458e8           # m/s
G = 6.67430e-11            # m^3 kg^-1 s^-2
hbar = 1.054571817e-34     # J·s
kB = 1.380649e-23          # J/K

H0 = 67.4 * 1e3 / 3.086e22 # Hubble rate [1/s]
Omega_m = 0.315
Omega_L = 0.685

L0 = 1e-12                 # STUR coherence length [m]
chi0 = 7.3e5               # torsion scale [1/s]

# ------------------------------------------------------------
# 2. 5D metric + evolution
# State = (t, x, y, z, chi)
# ------------------------------------------------------------
def scale_factor(t):
    """Simple FRW-like scale factor."""
    return np.exp(H0 * t)

def torsion_term(r, chi):
    """Toy torsion coupling."""
    return 1.0 + (L0 / (r + 1e-12)) * np.sin(chi)

def metric_5D(state):
    t, x, y, z, chi = state
    r = np.sqrt(x**2 + y**2 + z**2)
    a = scale_factor(t)
    T = torsion_term(r, chi)
    ds2 = -(c**2) * t**2 + a**2 * (x**2 + y**2 + z**2) + T * chi**2
    return ds2

def step_5D(state, dt):
    t, x, y, z, chi = state
    r = np.sqrt(x**2 + y**2 + z**2)

    # toy "forces" (effective geodesic drift)
    ax = -x / (1.0 + x**2 + chi**2)
    ay = -y / (1.0 + y**2 + chi**2)
    az = -z / (1.0 + z**2 + chi**2)

    chi_dot = chi0 * np.sin(0.1 * t)

    # Euler step (enough for a qualitative sandbox)
    t_new   = t + dt
    x_new   = x + ax * dt
    y_new   = y + ay * dt
    z_new   = z + az * dt
    chi_new = chi + chi_dot * dt

    return np.array([t_new, x_new, y_new, z_new, chi_new])

# ------------------------------------------------------------
# 3. Integrate a single worldline
# ------------------------------------------------------------
N_steps = 4000
dt = 0.002

state = np.array([0.0, 1.0, 0.8, 0.2, 0.1])
path = np.zeros((N_steps, 5))

for i in range(N_steps):
    path[i] = state
    state = step_5D(state, dt)

t_arr   = path[:, 0]
x_arr   = path[:, 1]
y_arr   = path[:, 2]
z_arr   = path[:, 3]
chi_arr = path[:, 4]

# ------------------------------------------------------------
# 4. Plots
# ------------------------------------------------------------
fig1, ax1 = plt.subplots()
ax1.plot(x_arr, y_arr, linewidth=1.0)
ax1.set_xlabel("x [arb]")
ax1.set_ylabel("y [arb]")
ax1.set_title("STUR 5D Worldline – x–y Projection")
ax1.grid(True)

fig2, ax2 = plt.subplots()
ax2.plot(np.arange(N_steps), chi_arr, linewidth=1.0)
ax2.set_xlabel("step")
ax2.set_ylabel("χ [arb]")
ax2.set_title("χ Torsion Evolution")
ax2.grid(True)

plt.show()

# ------------------------------------------------------------
# 5. Simple JSON summary
# ------------------------------------------------------------
summary = {
    "model": "STUR_5D_sandbox_v1",
    "N_steps": int(N_steps),
    "dt": float(dt),
    "t_final": float(t_arr[-1]),
    "x_span": [float(x_arr.min()), float(x_arr.max())],
    "y_span": [float(y_arr.min()), float(y_arr.max())],
    "chi_span": [float(chi_arr.min()), float(chi_arr.max())],
}

print("=== STUR 5D SANDBOX SUMMARY ===")
print(json.dumps(summary, indent=2))
