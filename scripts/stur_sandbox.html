STUR Sandbox

A compact, interactive sandbox for exploring a chosen geometry, its analytic derivative, and a minimal dynamical response. The goal is to generate exportable numerical artifacts suitable for downstream falsification and iteration.

S(u)=tanh(u)(1-e^{-u}) u=r/L₀ (dimensionless) Analytic dS/dr DPR-correct canvas JSON export

Purpose

This page provides a minimal, reproducible loop that connects: geometry → analytic derivative → toy dynamics → exportable data. It is intended as a scaffold for building and testing progressively higher-fidelity STUR simulations, including PDE evolutions, boundary-gate models, and higher-dimensional mappings.

\[ \boxed{ S(u)=\tanh(u)\left(1-e^{-u}\right), \quad u=\frac{r}{L_0} } \]

The demonstrator dynamics uses a simple radial rule driven by the geometry slope: \(F(r)=-\partial_r S(r)\), with linear damping and soft boundary reflections. This is a pedagogical baseline and is not presented as a complete physical model.

STUR 1D → 5D Starter Template (Copy Ready)

    

      This reference template reproduces the same dimensionless geometry \(u=r/L_0\), analytic derivative, and a clean JSON summary.
      It is compatible with JupyterLite / Pyodide environments (no animation blitting; optional JSON saving only).
    



# ===============================================================
# STUR 1D → 5D Manifold Suite — Pyodide/JupyterLite-safe
# Geometry: S(u)=tanh(u)(1-exp(-u)), u=r/L0
# ===============================================================
import numpy as np
import matplotlib.pyplot as plt
import json

# -------------------------------
# 1) Geometry (dimensionless u)
# -------------------------------
def Su(u):
    return np.tanh(u) * (1.0 - np.exp(-u))

def dSu_du(u):
    # d/du [tanh(u)(1-e^{-u})]
    sech2 = 1.0 / np.cosh(u)**2
    return sech2*(1.0 - np.exp(-u)) + np.tanh(u)*np.exp(-u)

def S_r(r, L0=1.0):
    u = r / L0
    return Su(u)

def dS_dr(r, L0=1.0):
    u = r / L0
    return dSu_du(u) / L0

def force(r, L0=1.0):
    return -dS_dr(r, L0)

# -------------------------------
# 2) Simple radial dynamics (demo)
# -------------------------------
def step(r, v, dt=0.02, L0=1.0, gamma=0.05, Rmax=5.0):
    F = force(r, L0)
    v = v + (F - gamma*v) * dt
    r = r + v * dt

    # soft wall bounce
    if r < 0.0:
        r = 0.0
        v *= -0.4
    if r > Rmax:
        r = Rmax
        v *= -0.4
    return r, v, F

# -------------------------------
# 3) Run 1D → 5D labeled suite
# -------------------------------
def simulate_dimension(D, L0=1.0, Rmax=5.0, steps=4000, dt=0.02, gamma=0.05):
    r = 0.6 * Rmax
    v = 0.0
    rs = np.zeros(steps)
    Fs = np.zeros(steps)
    for i in range(steps):
        r, v, F = step(r, v, dt=dt, L0=L0, gamma=gamma, Rmax=Rmax)
        rs[i] = r
        Fs[i] = F

    summary = {
        "D": int(D),
        "L0": float(L0),
        "Rmax": float(Rmax),
        "dt": float(dt),
        "gamma": float(gamma),
        "r_min": float(rs.min()),
        "r_max": float(rs.max()),
        "r_mean": float(rs.mean()),
        "F_rms": float(np.sqrt(np.mean(Fs**2))),
    }
    return rs, Fs, summary

def run_suite(max_dim=5, L0=1.0, Rmax=5.0, steps=4000, dt=0.02, gamma=0.05):
    trajs = {}
    summaries = []
    for D in range(1, max_dim+1):
        rs, Fs, s = simulate_dimension(D, L0=L0, Rmax=Rmax, steps=steps, dt=dt, gamma=gamma)
        trajs[D] = {"r": rs, "F": Fs}
        summaries.append(s)
    return trajs, summaries

# -------------------------------
# 4) Main
# -------------------------------
L0 = 1.0
Rmax = 5.0
dt = 0.02
gamma = 0.05
steps = 2500

trajs, summaries = run_suite(max_dim=5, L0=L0, Rmax=Rmax, steps=steps, dt=dt, gamma=gamma)

# Plot trajectories
fig, ax = plt.subplots(figsize=(8, 5))
t = np.arange(steps)
for D in trajs:
    ax.plot(t, trajs[D]["r"], label=f"{D}D")
ax.set_title("STUR 1D → 5D (same geometry, labeled dimensions)")
ax.set_xlabel("time step")
ax.set_ylabel("radius r")
ax.legend()
plt.tight_layout()
plt.show()

# Geometry curve
rgrid = np.linspace(0, Rmax, 600)
fig2, ax2 = plt.subplots(figsize=(8, 3.2))
ax2.plot(rgrid, S_r(rgrid, L0=L0))
ax2.set_title("Geometry S(r)")
ax2.set_xlabel("r")
ax2.set_ylabel("S")
plt.tight_layout()
plt.show()

# JSON summary (safe: works in JupyterLite; file save may depend on environment)
print("=== STUR 1D → 5D SUMMARY ===")
print(json.dumps(summaries, indent=2))
