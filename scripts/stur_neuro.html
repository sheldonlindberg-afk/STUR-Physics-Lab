#!/usr/bin/env python3
# ===============================================================
# STUR Neuroplasticity Lab - Exploratory Application

import numpy as np
import matplotlib.pyplot as plt

np.random.seed(0)

# ===============================================================
# STAGE 1 - GROWTH, COHERENCE, X-DIMENSION
# ===============================================================
mu, k_e, G_eff = 1.0, 1.0, 0.15
alpha_s, lambda_int, sigma_conf = 0.6, 3.0, 0.08
q1, q2, m1, m2 = -1.0, +1.0, 1.0, 1.0
gamma0, ell_B, L0 = 0.8, 1.0, 1.2

r_syn, tau_S = 6.0, 0.7
eta_w, lambda_w, sigma_r = 0.7, 0.02, 0.5 * 6.0
gamma_v, noise_amp = 1.0, 0.25

dt_g, steps_g = 0.02, 2200
t_g = np.arange(steps_g) * dt_g

d_neurons = 40.0
soma_A = np.array([-d_neurons/2, 0.0])
soma_B = np.array([+d_neurons/2, 0.0])
pos = soma_A + np.array([3.0, 0.0])
vel = np.zeros(2)

chi, k_chi, gamma_chi = 0.0, 1.0, 0.5

def S_reg(r):
    return np.tanh(r / L0) * (1.0 - np.exp(-r / L0))

def radial_bracket(r):
    r2 = r*r + 1e-9
    coul_grav = (k_e*q1*q2 - G_eff*m1*m2) / r2
    yukawa = alpha_s * ((1.0 - np.exp(-r/lambda_int)) / r2 + sigma_conf*r)
    return coul_grav + yukawa

def radial_force(r, c_SB):
    return S_reg(r) * radial_bracket(r) + (gamma0 / ell_B) * np.exp(-r/ell_B) * c_SB

def c_target(r):
    return np.tanh((r_syn - r) / (0.5 * r_syn))

xs, ys, rs, cs, ws, chi_hist = [], [], [], [], [], []
c_SB, w = -0.2, 0.0

for step in range(steps_g):
    diff = pos - soma_B
    r = max(np.linalg.norm(diff), 1e-6)
    r_hat = diff / r

    F_r = radial_force(r, c_SB)
    F_vec = F_r * r_hat

    acc = -F_vec/mu - gamma_v*vel
    vel += acc * dt_g + noise_amp * np.sqrt(dt_g) * np.random.randn(2)
    pos += vel * dt_g

    chi += (k_chi * (c_SB - 0.0) - gamma_chi * chi) * dt_g
    c_SB += (c_target(r) - c_SB) * dt_g / tau_S
    c_SB = float(np.clip(c_SB, -1.0, 1.0))

    proximity = np.exp(-(r - r_syn)**2 / (2.0 * sigma_r**2))
    gating = 0.5 * (1.0 + c_SB)
    w += (eta_w * proximity * gating * (1.0 - w) - lambda_w * (1.0 - gating) * w) * dt_g
    w = float(np.clip(w, 0.0, 1.0))

    xs.append(pos[0]); ys.append(pos[1])
    rs.append(r); cs.append(c_SB); ws.append(w); chi_hist.append(chi)

xs, ys, rs, cs, ws = map(np.array, (xs, ys, rs, cs, ws))
chi_hist = np.array(chi_hist)

r_final, c_final, w_final = float(rs[-1]), float(cs[-1]), float(ws[-1])
lambda_X = float(np.clip(0.5 * (1.0 + c_final) * w_final, 0.0, 1.5))

print("=== STUR Neuroplasticity ===")
print(f"r_final = {r_final:.3f} um")
print(f"c_final = {c_final:.3f}")
print(f"w_final = {w_final:.3f}")
print(f"lambda_X = {lambda_X:.3f}")

# ===============================================================
# STAGE 2 - HODGKIN-HUXLEY + STDP
# ===============================================================
C_m, g_Na, g_K, g_L = 1.0, 120.0, 36.0, 0.3
E_Na, E_K, E_L, E_syn = 50.0, -77.0, -54.4, 0.0

g0_syn = 1.0
g_syn_classic = g0_syn * w_final
g_syn_STUR = g0_syn * w_final * (1.0 + 2.0 * lambda_X)

factor_LTP_STUR = 1.0 + 4.0 * lambda_X
factor_LTD_STUR = max(0.0, 1.0 - 3.0 * lambda_X)

print(f"g_classic = {g_syn_classic:.3f}")
print(f"g_STUR = {g_syn_STUR:.3f}")

dt_hh, T_hh = 0.01, 120.0
steps_hh = int(T_hh / dt_hh)
t_hh = np.arange(steps_hh) * dt_hh

def I_ext_pre(t_ms):
    for t0 in [10.0, 40.0, 70.0, 100.0]:
        if t0 <= t_ms <= t0 + 3.0:
            return 10.0
    return 0.0

def alpha_n(V): return 0.01*(V + 55.0)/(1.0 - np.exp(-(V+55.0)/10.0))
def beta_n(V): return 0.125*np.exp(-(V+65.0)/80.0)
def alpha_m(V): return 0.1*(V + 40.0)/(1.0 - np.exp(-(V+40.0)/10.0))
def beta_m(V): return 4.0*np.exp(-(V+65.0)/18.0)
def alpha_h(V): return 0.07*np.exp(-(V+65.0)/20.0)
def beta_h(V): return 1.0/(1.0 + np.exp(-(V+35.0)/10.0))

V_pre, n_pre, m_pre, h_pre = -65.0, 0.3177, 0.0529, 0.5961
V_pc, n_pc, m_pc, h_pc = -65.0, 0.3177, 0.0529, 0.5961
V_ps, n_ps, m_ps, h_ps = -65.0, 0.3177, 0.0529, 0.5961
s_c, s_s = 0.0, 0.0
tau_syn, s_jump = 3.0, 0.6

V_pre_hist, V_pc_hist, V_ps_hist = np.zeros(steps_hh), np.zeros(steps_hh), np.zeros(steps_hh)
w_c_hist, w_s_hist = np.zeros(steps_hh), np.zeros(steps_hh)

w_c, w_s = w_final, w_final
A_plus, A_minus = 0.01, 0.012
tau_plus, tau_minus = 20.0, 20.0
decay_STDP, w_max, w_min = 1e-4, 2.0, 0.0

last_V_pre, last_V_post = V_pre, V_pc
last_pre_spike_time, last_post_spike_time = None, None
spike_thr = 0.0

def HH_step(V, n, m, h, I_inj):
    n += dt_hh * (alpha_n(V)*(1-n) - beta_n(V)*n)
    m += dt_hh * (alpha_m(V)*(1-m) - beta_m(V)*m)
    h += dt_hh * (alpha_h(V)*(1-h) - beta_h(V)*h)
    gNa, gK = g_Na * m**3 * h, g_K * n**4
    dV = (I_inj - gNa*(V-E_Na) - gK*(V-E_K) - g_L*(V-E_L)) / C_m
    return V + dt_hh*dV, n, m, h

for k in range(steps_hh):
    t_ms = t_hh[k]
    I_inj = I_ext_pre(t_ms) + 0.4 * np.random.randn()
    V_pre, n_pre, m_pre, h_pre = HH_step(V_pre, n_pre, m_pre, h_pre, I_inj)

    pre_spike = (last_V_pre < spike_thr and V_pre >= spike_thr)
    if pre_spike:
        s_c += s_jump
        s_s += s_jump
        if last_post_spike_time:
            dt = t_ms - last_post_spike_time
            if dt >= 0:
                w_c -= A_minus * np.exp(-dt / tau_minus)
                w_s -= A_minus * factor_LTD_STUR * np.exp(-dt / tau_minus)
        last_pre_spike_time = t_ms
    last_V_pre = V_pre

    s_c = max(0, s_c - s_c * dt_hh / tau_syn)
    s_s = max(0, s_s - s_s * dt_hh / tau_syn)

    g_syn_c = g_syn_classic * (w_c / max(w_final, 1e-6))
    I_syn_c = g_syn_c * s_c * (E_syn - V_pc)
    V_pc, n_pc, m_pc, h_pc = HH_step(V_pc, n_pc, m_pc, h_pc, I_syn_c)

    g_syn_s = g_syn_STUR * (w_s / max(w_final, 1e-6))
    I_syn_s = g_syn_s * s_s * (E_syn - V_ps)
    V_ps, n_ps, m_ps, h_ps = HH_step(V_ps, n_ps, m_ps, h_ps, I_syn_s)

    post_spike = (last_V_post < spike_thr and V_pc >= spike_thr)
    if post_spike:
        if last_pre_spike_time:
            dt = t_ms - last_pre_spike_time
            if dt >= 0:
                w_c += A_plus * np.exp(-dt / tau_plus)
                w_s += A_plus * factor_LTP_STUR * np.exp(-dt / tau_plus)
        last_post_spike_time = t_ms
    last_V_post = V_pc

    w_c = float(np.clip(w_c - decay_STDP*w_c*dt_hh, w_min, w_max))
    w_s = float(np.clip(w_s - decay_STDP*w_s*dt_hh, w_min, w_max))

    V_pre_hist[k], V_pc_hist[k], V_ps_hist[k] = V_pre, V_pc, V_ps
    w_c_hist[k], w_s_hist[k] = w_c, w_s

print(f"w_classic_final = {w_c_hist[-1]:.3f}")
print(f"w_STUR_final = {w_s_hist[-1]:.3f}")

# Plots
fig, axes = plt.subplots(2, 2, figsize=(12, 8))
fig.patch.set_facecolor('#050814')

for ax in axes.flatten():

axes[0,0].scatter(xs, ys, c=t_g, s=4, cmap="plasma")

axes[1,0].plot(t_hh, V_pre_hist, color='#60a5fa', lw=1, label='Pre')
axes[1,0].plot(t_hh, V_pc_hist, color='#f472b6', lw=1, label='Post (classic)')
axes[1,0].plot(t_hh, V_ps_hist, '--', color='#4ade80', lw=1, label='Post (STUR)')
axes[1,0].set_title("HH Voltages", color='#ffaa00')
axes[1,0].set_xlabel("Time (ms)", color='#94a3b8')

axes[1,1].plot(t_hh, w_s_hist, '--', color='#4ade80', lw=2, label='w_STUR')

axes[1,1].set_xlabel("Time (ms)", color='#94a3b8')
axes[1,1].legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')

fig.suptitle('STUR Neuroplasticity X-Synapse', color='#ffaa00', fontsize=14)
