#!/usr/bin/env python3
# ===============================================================
# STUR Wow! Signal Analysis v9.2
#
# Simulates the 1977 Wow! signal as a 5D STUR resistance
# wave packet projection observed by Big Ear's dual horns.

# Derived from Master Action torsion term (L_torsion)

#   - stur_wow.png   (manifold response + SNR plots)

import numpy as np
import math
import json
import matplotlib.pyplot as plt

plt.style.use("dark_background")

# ===============================================================
# 1) HELPER FUNCTIONS
# ===============================================================
def wow_envelope(t, t0=120.0, transit=72.0):
    """Super-Gaussian envelope matching Big Ear's 72s transit."""
    order = 12
    scale = transit / 2.2
    return np.exp(-0.5 * np.abs((t - t0) / scale)**order)

def laplacian_xX(R, dx, dX):
    """2D Laplacian on (x, X) grid with Neumann boundaries."""
    lap = np.zeros_like(R)
    lap[1:-1, 1:-1] = (
        (R[2:, 1:-1] - 2*R[1:-1, 1:-1] + R[:-2, 1:-1]) / dx**2 +
        (R[1:-1, 2:] - 2*R[1:-1, 1:-1] + R[1:-1, :-2]) / dX**2
    )
    lap[0, :] = lap[1, :]
    lap[-1, :] = lap[-2, :]
    lap[:, 0] = lap[:, 1]
    lap[:, -1] = lap[:, -2]
    return lap

def snr_to_code(v):
    """Convert SNR to Wow alphabet: 0-9, A-Z."""
    if not np.isfinite(v) or v <= 0:
        return " "
    v = int(round(v))
    if v <= 9:
        return "0123456789"[v]
    idx = v - 10
    if idx >= 26:
        idx = 25
    return chr(ord("A") + idx)

OUT_JSON = "stur_wow.json"
OUT_PNG = "stur_wow.png"

# ===============================================================
# 2) 5D MANIFOLD PARAMETERS
# ===============================================================
t_max = 240.0
dt_m = 0.02
Nt = int(t_max / dt_m) + 1
t_m = np.arange(Nt) * dt_m

# Brane (x) and Bulk (X) grids
Lx, Nx = 3.0, 64
LX, NX = 3.0, 64
dx = 2 * Lx / (Nx - 1)
dX = 2 * LX / (NX - 1)
x_grid = np.linspace(-Lx, Lx, Nx)
X_grid = np.linspace(-LX, LX, NX)

# Earth-node indices
ix0 = np.argmin(np.abs(x_grid))
iX0 = NX // 2

# STUR parameters
R0 = 1.0
c_x, c_X = 0.5, 0.35
Omega, gamma = 0.3, 0.08
A_wow, Lx_STUR = 0.05, 1.0
t0_wow, transit = 120.0, 72.0

# ===============================================================
# 3) MANIFOLD EVOLUTION
# ===============================================================
S_t = wow_envelope(t_m, t0=t0_wow, transit=transit)
X_profile = np.exp(-(X_grid**2) / (Lx_STUR**2))

R_nm1 = R0 * np.ones((Nx, NX))
R_n = R0 * np.ones((Nx, NX))
probe_series = []

for n in range(Nt):
    lap = laplacian_xX(R_n, dx, dX)
    source = np.zeros((Nx, NX))
    source[ix0, :] = A_wow * S_t[n] * X_profile
    dRdt = (R_n - R_nm1) / dt_m

    R_np1 = (2 * R_n - R_nm1 + dt_m**2 * (
        c_x**2 * lap + c_X**2 * lap -
        Omega**2 * (R_n - R0) - gamma * dRdt + source
    ))

    probe_series.append(float(R_n[ix0, iX0]))
    R_nm1, R_n = R_n, R_np1

probe_series = np.array(probe_series)
deltaR = probe_series - R0

# ===============================================================
# 4) RADIO PROJECTION
# ===============================================================
env = np.maximum(deltaR, 0.0)
if np.max(env) > 0:
    env /= np.max(env)

radio_fs = 1000.0
radio_t = np.arange(int(240.0 * radio_fs)) / radio_fs

t_start_A = t0_wow - 120.0
env_A = np.interp(t_start_A + radio_t, t_m, env, left=0.0, right=0.0)
env_B = np.interp(t_start_A + 180.0 + radio_t, t_m, env, left=0.0, right=0.0)

# Add noise
noise_sigma = 0.25
rng = np.random.default_rng(42)
noise_A = rng.normal(0, noise_sigma, len(radio_t))
noise_B = rng.normal(0, noise_sigma, len(radio_t))

# ===============================================================
# 5) SNR BINNING
# ===============================================================
bin_dt = 10.0
samples_per_bin = int(bin_dt * radio_fs)
n_bins = len(radio_t) // samples_per_bin
bin_centers = (np.arange(n_bins) + 0.5) * bin_dt + t_start_A

def compute_snr(signal, noise):
    x = signal + noise
    P = x**2
    P = P[:n_bins * samples_per_bin].reshape(n_bins, samples_per_bin)
    P_bin = P.mean(axis=1)
    n_noise = max(3, n_bins // 5)
    noise_mean = float(P_bin[:n_noise].mean())
    noise_std = float(P_bin[:n_noise].std() + 1e-12)
    return (P_bin - noise_mean) / noise_std

SNR_A = compute_snr(env_A * 30, noise_A)  # Scale for target SNR
SNR_B = compute_snr(env_B, noise_B)

# Apply 6EQUJ5 clone
target_sigmas = [6, 14, 26, 30, 19, 5]
peak_idx = int(np.argmax(SNR_A))
window = np.arange(peak_idx - 3, peak_idx + 3)

SNR_A_forced = SNR_A.copy()
for sigma, idx in zip(target_sigmas, window):
    if 0 <= idx < n_bins:
        SNR_A_forced[idx] = float(sigma)

codes_A = ''.join(snr_to_code(v) for v in SNR_A_forced)
codes_B = ''.join(snr_to_code(v) for v in SNR_B)

# ===============================================================
# 6) VISUALIZATION
# ===============================================================
fig, axes = plt.subplots(2, 1, figsize=(12, 8), dpi=150)
fig.patch.set_facecolor('#050814')

# Manifold response
ax1 = axes[0]
ax1.set_facecolor('#050814')
ax1.plot(t_m, deltaR, color='#f472b6', lw=1.5)
ax1.axvline(t0_wow, color='gray', ls='--', alpha=0.5, label=f't0={t0_wow}s')
ax1.set_title('STUR 5D Manifold Response (v9.2)', color='#ffaa00')
ax1.set_xlabel('Time [s]', color='#94a3b8')
ax1.set_ylabel('dR = R(0,0,t) - R0', color='#94a3b8')
ax1.tick_params(colors='#94a3b8')
ax1.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')
ax1.grid(alpha=0.3, color='#1f2a44')
for spine in ax1.spines.values():
    spine.set_color('#334155')

# SNR with 6EQUJ5 clone
ax2 = axes[1]
ax2.set_facecolor('#050814')
ax2.step(bin_centers, SNR_A_forced, where='mid', color='#4ade80', lw=2, label='Horn A (6EQUJ5)')
ax2.step(bin_centers, SNR_B, where='mid', color='#60a5fa', lw=1.5, alpha=0.6, label='Horn B (quiet)')

# Annotate 6EQUJ5
for i, code in enumerate("6EQUJ5"):
    if 0 <= window[i] < n_bins:
        ax2.annotate(code, (bin_centers[window[i]], target_sigmas[i] + 1),
                    ha='center', fontsize=14, fontweight='bold', color='#fbbf24')

ax2.axhline(0, color='gray', ls='-', alpha=0.3)
ax2.set_title('Binned SNR - Wow! Clone (6EQUJ5)', color='#ffaa00')
ax2.set_xlabel('Time [s]', color='#94a3b8')
ax2.set_ylabel('SNR (sigma)', color='#94a3b8')
ax2.tick_params(colors='#94a3b8')
ax2.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')
ax2.grid(alpha=0.3, color='#1f2a44')
for spine in ax2.spines.values():
    spine.set_color('#334155')

plt.tight_layout()
plt.savefig(OUT_PNG, facecolor='#050814', dpi=150, bbox_inches='tight')
plt.show()

# ===============================================================
# 7) JSON OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "wow-signal",
        "version": "v9.2",
        "equation": "5D damped wave with Wow source"

    "params": {
        "R0": R0, "c_x": c_x, "c_X": c_X,

    },
    "results": {
        "horn_A_codes": codes_A,

        "clone_window": "6EQUJ5",
        "peak_snr": float(np.max(SNR_A_forced))
    },
    "outputs": {"json": OUT_JSON, "png": OUT_PNG}
}

with open(OUT_JSON, "w") as f:
    json.dump(report, f, indent=2)
