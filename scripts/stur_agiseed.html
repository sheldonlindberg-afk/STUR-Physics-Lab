#!/usr/bin/env python3
# ===============================================================
# STUR Double Slit — DATASET-LOCKED PLASMON COHERENCE (CANONICAL)
# Pure Python • iOS-safe • No external dependencies
#
# Output:
#   - stur_plasmon_datasetlocked.json
#   - stur_plasmon_datasetlocked.svg
#
# Notes:
# - Dataset-locked plasmon parameters -> coherence length L0(λ)
# - QM baseline = Fraunhofer double slit (envelope * fringes)
# - STUR = X-bundle phase dispersion (numeric + analytic cross-check)
# - Falsified if RMSE_STUR >= RMSE_QM
# ===============================================================

import math, json, time

# ===============================================================
# 1) VERIFIED PLASMON PARAMETERS (DATASET-LOCKED)
# ===============================================================
LAMBDA_RES_NM = 532.0
HALFMAX_L_NM  = 493.7
HALFMAX_R_NM  = 575.5
N_EFF         = 1.50

OUT_JSON = "stur_plasmon_datasetlocked.json"
OUT_SVG  = "stur_plasmon_datasetlocked.svg"

# ===============================================================
# 2) EXPERIMENTAL PROXY DATA (VISIBILITY vs WAVELENGTH)
# ===============================================================
WL_NM = [350,360,380,400,420,440,450,460,470,480,500,520,540,560,580,600,620,640]
V_DATA = [0.98,0.97,0.93,0.86,0.79,0.71,0.66,0.62,0.59,0.57,0.55,0.57,0.60,0.64,0.69,0.75,0.82,0.88]
RQ_EFF = [0.18,0.20,0.24,0.30,0.38,0.52,0.60,0.58,0.55,0.53,0.50,0.47,0.44,0.40,0.36,0.32,0.28,0.25]

# ===============================================================
# 3) NUMERICAL UTILITIES
# ===============================================================
def trapz(f, x):
    s = 0.0
    for i in range(len(x)-1):
        s += 0.5*(f[i]+f[i+1])*(x[i+1]-x[i])
    return s

def normalize(f, x):
    z = trapz(f, x)
    if (not math.isfinite(z)) or abs(z) < 1e-30:
        return list(f)
    return [v/z for v in f]

def rmse(a, b):
    n = min(len(a), len(b))
    return math.sqrt(sum((a[i]-b[i])**2 for i in range(n))/n)

def sinc(x):
    return 1.0 if abs(x) < 1e-12 else math.sin(x)/x

def clamp01(v):
    return 0.0 if v < 0.0 else (1.0 if v > 1.0 else v)

# ===============================================================
# 4) PLASMON → COHERENCE LENGTH
# ===============================================================
def plasmon_Q():
    fwhm = max(HALFMAX_R_NM - HALFMAX_L_NM, 1e-12)
    return LAMBDA_RES_NM / fwhm

Q_PLASMON = plasmon_Q()
FWHM_NM   = HALFMAX_R_NM - HALFMAX_L_NM

def L0_from_plasmon(lam_m):
    # dataset-locked mapping: L0(λ) = (Q / (2π n_eff)) λ
    return (Q_PLASMON / (2.0*math.pi*N_EFF)) * lam_m

# ===============================================================
# 5) DOUBLE-SLIT GEOMETRY
# ===============================================================
d = 1.0e-3      # slit separation [m]
a = 0.15e-3     # slit width [m]
L = 5.0         # screen distance [m]

Ny = 801
y  = [-0.03 + 0.06*i/(Ny-1) for i in range(Ny)]

# ===============================================================
# 6) VISIBILITY EXTRACTION (robust, windowed, env-div safe)
# ===============================================================
def local_visibility(I, env, window=0.006):
    J = []
    for i in range(len(y)):
        if abs(y[i]) <= window:
            J.append(I[i] / max(env[i], 1e-30))

    if len(J) < 5:
        return 0.0

    peaks, troughs = [], []
    for i in range(1, len(J)-1):
        if J[i] > J[i-1] and J[i] > J[i+1]:
            peaks.append(J[i])
        if J[i] < J[i-1] and J[i] < J[i+1]:
            troughs.append(J[i])

    if (not peaks) or (not troughs):
        return 0.0

    m = min(len(peaks), len(troughs))
    s = 0.0
    for i in range(m):
        s += (peaks[i]-troughs[i]) / (peaks[i]+troughs[i] + 1e-30)
    return s/m

# ===============================================================
# 7) QM BASELINE (NO DEPHASING)
# ===============================================================
def qm_visibility(lam):
    env = [sinc(a*(yy/L)/lam)**2 for yy in y]
    fr  = [math.cos(math.pi*d*(yy/L)/lam)**2 for yy in y]
    I   = normalize([env[i]*fr[i] for i in range(len(y))], y)
    return local_visibility(I, env)

# ===============================================================
# 8) STUR PHASE MODULATOR
# ===============================================================
def S_u(u):
    return math.tanh(u) * (1.0 - math.exp(-u))

def stur_visibility_numeric(lam, Rq):
    L0 = L0_from_plasmon(lam)

    NX   = 401
    Xmax = 5.0*L0
    X    = [-Xmax + 2.0*Xmax*i/(NX-1) for i in range(NX)]
    w    = normalize([math.exp(-0.5*(xx/L0)**2) for xx in X], X)

    env = [sinc(a*(yy/L)/lam)**2 for yy in y]
    phi = [(2.0*math.pi/lam) * d*yy/L for yy in y]

    I = []
    for i, yy in enumerate(y):
        path = abs(d*yy/L)
        Su   = S_u(path / max(L0, 1e-30))

        acc = 0.0
        for j, xx in enumerate(X):
            chi = Rq * Su * (xx / max(L0, 1e-30))
            A   = 2.0*math.sqrt(env[i]) * math.cos(0.5*(phi[i] + chi))
            acc += w[j] * (A*A)
        I.append(acc)

    I = normalize(I, y)
    return local_visibility(I, env)

def stur_visibility_analytic(lam, Rq, window=0.006):
    L0 = L0_from_plasmon(lam)

    su2 = []
    for yy in y:
        if abs(yy) <= window:
            path = abs(d*yy/L)
            su2.append(S_u(path / max(L0,1e-30))**2)

    if not su2:
        return 0.0

    mean_su2 = sum(su2)/len(su2)
    return math.exp(-0.125 * (Rq**2) * mean_su2)

# ===============================================================
# 9) SVG PLOT (no matplotlib)
# ===============================================================
def write_svg(path, xs, series, labels, title):
    W, H, pad = 1200, 600, 70
    x0, x1 = min(xs), max(xs)

    def Xp(x):
        return pad + (W-2*pad)*(x-x0)/(x1-x0)

    def Yp(v):
        return H-pad - (H-2*pad)*clamp01(v)

    svg = []
    svg.append(f'')
    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(svg))

# ===============================================================
# 10) RUN
# ===============================================================
V_QM, V_STUR, V_STUR_A, L0_M = [], [], [], []

for wl, rq in zip(WL_NM, RQ_EFF):
    lam = wl * 1e-9
    L0_M.append(L0_from_plasmon(lam))
    V_QM.append(qm_visibility(lam))
    V_STUR.append(stur_visibility_numeric(lam, rq))
    V_STUR_A.append(stur_visibility_analytic(lam, rq))

rmse_qm = rmse(V_QM, V_DATA)
rmse_st = rmse(V_STUR, V_DATA)

write_svg(
    OUT_SVG,
    WL_NM,
    [V_DATA, V_QM, V_STUR, V_STUR_A],
    ["DATA", "QM", "STUR-num", "STUR-analytic"],
    "QM vs STUR — plasmon-derived L0(λ), dataset-locked"
)

report = {
  "timestamp_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
  "plasmon": {
    "lambda_res_nm": LAMBDA_RES_NM,
    "halfmax_L_nm": HALFMAX_L_NM,
    "halfmax_R_nm": HALFMAX_R_NM,
    "fwhm_nm": FWHM_NM,
    "Q": Q_PLASMON,
    "n_eff": N_EFF
  },
  "geometry": {"d_m": d, "a_m": a, "L_m": L, "window_m": 0.006},
  "dataset": {"wl_nm": WL_NM, "V_data": V_DATA, "Rq_eff": RQ_EFF},
  "derived": {
    "L0_m": L0_M,
    "V_qm": V_QM,
    "V_stur": V_STUR,
    "V_stur_analytic": V_STUR_A,
    "residuals_qm": [V_QM[i]-V_DATA[i] for i in range(len(WL_NM))],
    "residuals_stur": [V_STUR[i]-V_DATA[i] for i in range(len(WL_NM))]
  },
  "rmse": {"QM": rmse_qm, "STUR": rmse_st, "delta": rmse_st - rmse_qm},
  "falsified": (rmse_st >= rmse_qm),
  "outputs": {"json": OUT_JSON, "svg": OUT_SVG},
  "citations": [
    "Born & Wolf — Principles of Optics",
    "Goodman — Statistical Optics",
    "Au nanoparticle plasmon linewidth spectroscopy (dataset-locked)"
  ]
}

with open(OUT_JSON, "w", encoding="utf-8") as f:
    json.dump(report, f, indent=2)

print("=== STUR DATASET-LOCKED RUN ===")
print("Q_plasmon =", round(Q_PLASMON, 6))
print("FWHM_nm   =", round(FWHM_NM, 3))
print("RMSE(QM)  =", round(rmse_qm, 8))
print("RMSE(STUR)=", round(rmse_st, 8))
print("ΔRMSE     =", round(rmse_st - rmse_qm, 8))
print("FALSIFIED =", report["falsified"])
print("Wrote:", OUT_SVG, "and", OUT_JSON)
