#!/usr/bin/env python3
"""
STUR QC Lattice — ∞₃ Geometric Framework
==================================================
Tests STUR-modulated quantum evolution on 1D lattice.

Complete EFT Framework:
  - 4 inputs: M_Planck, v_EW, m_t, alpha_em (lattice params L0/N/J/sigma_R below are
  #   simulation inputs, not derived from these — see disclosure in prose)
  - ∞₃ geometric framework
  - Quantum behavior emerges from ∞-helix geometry

Derives from: ∞-helix topology → discrete phases → R_STUR generator → i hbar dPsi/dt = R_STUR Psi
"""

import numpy as np
import math
import json
import time
import matplotlib.pyplot as plt
from numpy.linalg import eig

plt.style.use("dark_background")

# ===============================================================
# STUR MODULATOR [emerges from ∞-helix geometry]
# S(r) = tanh(r/L₀) · (1 - exp(-r/L₀))
# L₀ derived from M_Planck — geometric framework
# ===============================================================
def S_STUR(r, L0):
    x = r / L0
    return math.tanh(x) * (1.0 - math.exp(-x))

# Pauli matrices
I2 = np.eye(2, dtype=complex)
sx = np.array([[0,1],[1,0]], dtype=complex)
sy = np.array([[0,-1j],[1j,0]], dtype=complex)
sz = np.array([[1,0],[0,-1]], dtype=complex)
s_plus = np.array([[0,1],[0,0]], dtype=complex)
s_minus = np.array([[0,0],[1,0]], dtype=complex)

def kron_all(ops):
    out = ops[0]
    for op in ops[1:]:
        out = np.kron(out, op)
    return out

def single_op(pauli, site, N):
    ops = [I2 for _ in range(N)]
    ops[site] = pauli
    return kron_all(ops)

def two_op(pauli_a, i, pauli_b, j, N):
    ops = [I2 for _ in range(N)]
    ops[i] = pauli_a
    ops[j] = pauli_b
    return kron_all(ops)

# ===============================================================
# STUR RESISTANCE GENERATOR
# ===============================================================
def build_R_STUR(N, L0=1.0, a_lat=1.0, h0=1.0, J0=1.0, Omega=0.5):
    dim = 2**N
    R = np.zeros((dim, dim), dtype=complex)
    positions = np.array([i * a_lat for i in range(N)], dtype=float)

    # On-site terms: h0 * S(r_i) * sigma_z
    for i in range(N):
        S_i = S_STUR(positions[i], L0)
        R += h0 * S_i * single_op(sz, i, N)

    # XX+YY couplings: J_ij * S(r_ij) * (sigma_x sigma_x + sigma_y sigma_y)
    for i in range(N):
        for j in range(i+1, N):
            r_ij = abs(positions[i] - positions[j])
            S_ij = S_STUR(r_ij, L0)
            J_ij = J0 * S_ij
            R += J_ij * (two_op(sx, i, sx, j, N) + two_op(sy, i, sy, j, N))

    # Transverse drive: Omega * S(r_i) * sigma_x
    for i in range(N):
        S_i = S_STUR(positions[i], L0)
        R += Omega * S_i * single_op(sx, i, N)

    return R, positions

# ===============================================================
# DETERMINISTIC EVOLUTION
# ===============================================================
def evolve_STUR(R, psi0, t_final, n_steps, hbar=1.0):
    dt = t_final / n_steps
    w, V = np.linalg.eigh(R)
    phase = np.exp(-1j * w * dt / hbar)

    psi = psi0.astype(complex)
    traj = [psi.copy()]

    for _ in range(n_steps):
        c = V.conj().T @ psi
        c *= phase
        psi = V @ c
        psi /= np.linalg.norm(psi)
        traj.append(psi.copy())

    return np.array(traj), dt

# ===============================================================
# ENTANGLEMENT ENTROPY
# ===============================================================
def entanglement_entropy(psi, N, NA):
    NB = N - NA
    dimA, dimB = 2**NA, 2**NB
    psi_mat = psi.reshape(dimA, dimB)
    rho_A = psi_mat @ psi_mat.conj().T
    vals = np.real_if_close(np.linalg.eigvalsh(rho_A))
    vals = np.clip(vals, 0.0, 1.0)
    lam = vals[vals > 1e-12]
    return float(-np.sum(lam * np.log2(lam))) if lam.size > 0 else 0.0

def resistance_flux(psi, R):
    return float(np.real(np.vdot(psi, R @ psi)))

# ===============================================================
# LOCAL DENSITIES AND CURRENTS
# ===============================================================
def local_density_ops(N):
    return [0.5 * (single_op(I2, i, N) - single_op(sz, i, N)) for i in range(N)]

def local_densities(traj, n_ops):
    return np.array([[float(np.real(np.vdot(psi, op @ psi))) for op in n_ops] for psi in traj])

def continuity_max_residual(dens, currents, dt):
    T, N = dens.shape
    max_res = 0.0
    for t in range(1, T-1):
        d_rho = (dens[t+1] - dens[t-1]) / (2.0 * dt)
        for i in range(N):
            res = abs(d_rho[i] + np.sum(currents[t][i,:]))
            max_res = max(max_res, res)
    return max_res

# ===============================================================
# MAIN SIMULATION
# ===============================================================
N = 6
NA = 3
L0 = 1.0
h0 = 1.0
J0 = 0.8
Omega = 0.6
t_final = 5.0
n_steps = 200
dim = 2**N

print(f"=== STUR QC Lattice ===")
print(f"Building R_STUR for N = {N}, dim = {dim}")

R_full, positions = build_R_STUR(N, L0, 1.0, h0, J0, Omega)

psi0 = np.zeros(dim, dtype=complex)
psi0[0] = 1.0

print("Evolving deterministic STUR field...")
traj_full, dt_full = evolve_STUR(R_full, psi0, t_final, n_steps)

S_vals = [entanglement_entropy(psi, N, NA) for psi in traj_full]
flux_vals = [resistance_flux(psi, R_full) for psi in traj_full]

S_max = max(S_vals)
flux0 = flux_vals[0]
flux_drift = (max(flux_vals) - min(flux_vals)) / (abs(flux0) + 1e-15)

# ===============================================================
# FALSIFICATION CHECK

excl1 = False  # Would need continuity run
