Falsification Criteria

STUR proto-cell predictions are falsified if any of the following are observed:

Kill 1 — No Bifurcation Threshold

If structure formation scales linearly with catalytic activity (no critical threshold), STUR's bifurcation prediction LIFE.1 is falsified.

Kill 2 — Order Parameter Never Spikes

If the variance/mean ratio \(\Quant{O_{\rm life}}\) remains below 3 during all structure formation events, STUR's order parameter prediction LIFE.2 is falsified.

Kill 3 — No Pore Size Optimum

If structure formation is equally likely across all pore sizes (5–50 µm) without an optimum, STUR's geometry prediction LIFE.3 is falsified.

Kill 4 — Structure at Extreme Temperatures

If proto-cells form readily at T < 70°C or T > 130°C, STUR's temperature window prediction LIFE.4 is falsified.

Experimental Protocol: Microfluidic vent simulators with variable pore geometry, temperature gradient, and FeS/NiS surface density. Monitor acetyl-thioester localization via fluorescent probes.

Executable Simulation

STUR hydrothermal vent proto-cell simulation with curvature-driven life emergence. Copy and run in any Python 3 environment.

#!/usr/bin/env python3
# ===============================================================
# STUR Hydrothermal Vent Proto-Cell
#
# Alkaline vent micro-pore with STUR curvature selection:
#   - 12 um pore, 80-120 C gradient, pH 9.8 -> 6.0
#   - FeS/NiS-like wall catalysis
#   - Thermophoresis for acetyl-thioester
#   - X-feedback f_X = exp[chi(R - R_bg)]
#
# Derived from Master Action X-coupling term
# ===============================================================

import numpy as np
import json
import matplotlib.pyplot as plt

# Grid and geometry
N = 32
L = 12e-6
dx = L / N
x = np.linspace(-L/2, L/2, N, endpoint=False)
X, Y, Z = np.meshgrid(x, x, x, indexing="ij")

# Temperature field (80-120 C)
T_cold = 353.15
T_hot = 393.15
T = T_cold + (T_hot - T_cold)*((Z + L/2)/L)
gradT = (T_hot - T_cold)/L

# pH field (9.8 -> 6.0)
pH_top = 9.8
pH_bottom = 6.0
pH = pH_top + (pH_bottom - pH_top)*((Z + L/2)/L)

# Initial chemistry (mol/m^3)
CO2 = np.full((N, N, N), 15.0)
H2 = np.full((N, N, N), 1.0)
Ac = np.zeros((N, N, N))
ATP = np.full((N, N, N), 1e-3)
Fe2 = np.full((N, N, N), 1.0)

# Diffusion coefficients
D_CO2 = 1.5e-9
D_H2 = 4.0e-9
D_Ac = 2.0e-10
D_ATP = 1.2e-10

# Thermophoresis
S_T_Ac = 0.55
v_th = -D_Ac * S_T_Ac * gradT / T

# FeS/NiS catalysis
Ea = 68e3
R_gas = 8.314
k_base = 5.0
kT = k_base * np.exp(-Ea / R_gas * (1.0/T - 1.0/373.15))
r_cyl = np.sqrt(X*X + Y*Y)
dist_from_wall = (L/2 - r_cyl)
surface = 1.0 + 30.0 * np.exp(-(dist_from_wall/(0.6e-6))**2)

# STUR resistance weights
w_CO2, w_H2, w_Ac, w_ATP, w_Fe = 0.1, 0.05, 10.0, 20.0, 0.2
eps = 1e-35

# Time stepping
dt = 0.25 * dx*dx / max(D_CO2, D_H2, D_Ac, D_ATP)
nt = 15000

def lap(field):
    field = np.clip(field, 0.0, 1e6)
    return (np.roll(field,1,0)+np.roll(field,-1,0)+
            np.roll(field,1,1)+np.roll(field,-1,1)+
            np.roll(field,1,2)+np.roll(field,-1,2) - 6.0*field) / (dx*dx)

def advect_z(field, v_z, dx, dt):
    field = np.clip(field, 0.0, 1e6)
    c_up = np.roll(field, 1, axis=2)
    c_down = np.roll(field, -1, axis=2)
    c_from = np.where(v_z > 0.0, c_up, c_down)
    grad = np.clip((field - c_from) / dx, -1e9, 1e9)
    return np.clip(field - dt * v_z * grad, 0.0, 1e6)

# Run simulation with chi_X = 4.0
chi = 4.0
CO2_f, H2_f, Ac_f, ATP_f, Fe_f = CO2.copy(), H2.copy(), Ac.copy(), ATP.copy(), Fe2.copy()
history = {"t": [], "Ac_mean": [], "ATP_mean": [], "R_var": [], "O_life": []}

for n in range(nt):
    R = w_CO2*CO2_f + w_H2*H2_f + w_Ac*Ac_f + w_ATP*ATP_f + w_Fe*Fe_f
    R = np.clip(R, 0.0, 1e6)
    Rbg = R.mean()
    fX = np.exp(np.clip(chi*(R - Rbg), -30.0, 30.0))

    rate = np.minimum(surface * kT * CO2_f * H2_f * (Ac_f + 1e-6) * fX, 5e3)

    CO2_f = np.clip(CO2_f + dt*(D_CO2*lap(CO2_f) - rate), 0.0, 1e6)
    H2_f = np.clip(H2_f + dt*(D_H2*lap(H2_f) - rate), 0.0, 1e6)
    Ac_f = np.clip(Ac_f + dt*(D_Ac*lap(Ac_f) + rate - 0.01*Ac_f), 0.0, 1e6)
    ATP_f = np.clip(ATP_f + dt*(D_ATP*lap(ATP_f) + 0.4*rate), 0.0, 1e6)

    Ac_f = Ac_f / (1.0 + dt*0.01)
    Ac_f = advect_z(Ac_f, v_th, dx, dt)

    CO2_f *= 0.999999
    H2_f *= 0.999999
    Ac_f *= 0.999999
    ATP_f *= 0.999999

    if n % 2000 == 0:
        t_now = n*dt
        Ac_mean = Ac_f.mean()
        R_var = ((R - Rbg)**2).mean()
        O_life = R_var / (Ac_mean + 1e-12)
        history["t"].append(t_now)
        history["Ac_mean"].append(Ac_mean)
        history["ATP_mean"].append(ATP_f.mean())
        history["R_var"].append(R_var)
        history["O_life"].append(O_life)

print("=== STUR Hydrothermal Proto-Cell ===")
print(f"chi_X = {chi}")
print(f"Final Ac_mean = {Ac_f.mean():.4e}")
print(f"Final ATP_mean = {ATP_f.mean():.4e}")
print(f"Max R_var = {max(history['R_var']):.3e}")

# Plots
fig, axes = plt.subplots(2, 2, figsize=(12, 8))
fig.patch.set_facecolor('#050814')

for ax in axes.flatten():
    ax.set_facecolor('#050814')
    ax.tick_params(colors='#94a3b8')
    ax.grid(True, alpha=0.3, color='#1f2a44')
    for spine in ax.spines.values():
        spine.set_color('#334155')

t = np.array(history["t"])
axes[0,0].plot(t, history["Ac_mean"], color='#4ade80', lw=2)
axes[0,0].set_title("Acetyl Mean", color='#ffaa00')
axes[0,0].set_ylabel("[Ac] mean", color='#94a3b8')

axes[0,1].plot(t, history["ATP_mean"], color='#60a5fa', lw=2)
axes[0,1].set_title("ATP Mean", color='#ffaa00')
axes[0,1].set_ylabel("[ATP] mean", color='#94a3b8')

axes[1,0].plot(t, history["R_var"], color='#f472b6', lw=2)
axes[1,0].set_title("Resistance Variance", color='#ffaa00')
axes[1,0].set_xlabel("Time (s)", color='#94a3b8')
axes[1,0].set_ylabel("Var(R)", color='#94a3b8')

axes[1,1].plot(t, history["O_life"], color='#fbbf24', lw=2)
axes[1,1].set_title("Life-Onset Order Parameter", color='#ffaa00')
axes[1,1].set_xlabel("Time (s)", color='#94a3b8')
axes[1,1].set_ylabel("O_life", color='#94a3b8')

fig.suptitle(f'STUR Hydrothermal Proto-Cell (chi_X={chi})', color='#ffaa00', fontsize=14)
plt.tight_layout()
plt.show()
