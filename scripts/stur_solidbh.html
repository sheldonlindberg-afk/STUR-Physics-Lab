# ===============================================================
# 1) FUNDAMENTAL CONSTANTS (SI units)
# ===============================================================
@dataclass(frozen=True)
class Constants:
    """ONE input: M_Planck. All else derived."""
    hbar: float = 1.054571817e-34   # J⋅s
    c: float = 2.99792458e8         # m/s
    G: float = 6.67430e-11          # m³/(kg⋅s²)
    k_B: float = 1.380649e-23       # J/K
    M_sun: float = 1.989e30         # kg

    @property
    def M_Pl(self):
        """Planck mass — THE single input"""
        return np.sqrt(self.hbar * self.c / self.G)

    @property
    def l_Pl(self):
        """Planck length — derived from M_Pl"""
        return np.sqrt(self.hbar * self.G / self.c**3)

    @property
    def t_Pl(self):
        """Planck time — derived from M_Pl"""
        return np.sqrt(self.hbar * self.G / self.c**5)

# ===============================================================
# 2) BLACK HOLE THERMODYNAMICS (geometric framework)
# ===============================================================
def schwarzschild_radius(M, C):
    """r_s = 2GM/c² — from TEGR torsion boundary"""
    return 2 * C.G * M / C.c**2

def surface_gravity(M, C):
    """κ = c⁴/(4GM) — from TEGR horizon structure"""
    return C.c**4 / (4 * C.G * M)

def hawking_temperature(M, C):
    """T_H = ℏκ/(2πck_B) — derived from TEGR"""
    kappa = surface_gravity(M, C)
    return C.hbar * kappa / (2 * np.pi * C.c * C.k_B)

def bekenstein_entropy(M, C):
    """S_BH = k_B⋅A/(4ℓ_Pl²) — holographic from ∞₃"""
    r_s = schwarzschild_radius(M, C)
    A = 4 * np.pi * r_s**2
    return C.k_B * A / (4 * C.l_Pl**2)

def hawking_power(M, C):
    """P = ℏc⁶/(15360πG²M²) — radiation power"""
    return C.hbar * C.c**6 / (15360 * np.pi * C.G**2 * M**2)

def evaporation_time(M, C):
    """t_evap = 5120πG²M³/(ℏc⁴) — complete evaporation"""
    return 5120 * np.pi * C.G**2 * M**3 / (C.hbar * C.c**4)
