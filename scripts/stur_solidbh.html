# ===============================================================
# 1) PARAMETERS
# ===============================================================
@dataclass(frozen=True)
class Params:
    R_bg: float = 0.0
    m: float = 0.45
    rho_c: float = 1.0
    H_sign_init: int = -1
    eps: float = 1e-12

def V(R, p):
    return 0.5 * (p.m ** 2) * (R - p.R_bg) ** 2

def dVdR(R, p):
    return (p.m ** 2) * (R - p.R_bg)

def rho_R(R, Rdot, p):
    return 0.5 * (Rdot ** 2) + V(R, p)

def rho_eff(rho, p):
    if rho <= 0:
        return 0.0
    r = rho / (p.rho_c + p.eps)
    val = rho * (1.0 - r)
    return float(max(0.0, val))

# ===============================================================
# 2) RK4 INTEGRATOR
# ===============================================================
def stur_system(t, y, p):
    a, R, Rdot, Hs = y
    H_sign = -1 if Hs < 0 else +1

    rho = rho_R(R, Rdot, p)
    heff = rho_eff(rho, p)
    H = float(H_sign) * np.sqrt(heff / 3.0 + p.eps)

    near_bounce = (abs(H) < 5e-4) and (rho > 0.92 * p.rho_c)
    if near_bounce and H_sign < 0:
        H_sign = +1

    adot = a * H
    dR = Rdot
    dRdot = -3.0 * H * Rdot - dVdR(R, p)
    dHs = 0.0

    return np.array([adot, dR, dRdot, dHs], dtype=float)

def rk4_integrate(p, t_span, n, a0, R0, Rdot0):
    t = np.linspace(t_span[0], t_span[1], n)
    dt = t[1] - t[0]
    y = np.zeros((4, n), dtype=float)
    y[:, 0] = [a0, R0, Rdot0, float(-1 if p.H_sign_init < 0 else +1)]

    for i in range(n - 1):
        ti = t[i]
        yi = y[:, i]
        k1 = stur_system(ti, yi, p)
        k2 = stur_system(ti + 0.5 * dt, yi + 0.5 * dt * k1, p)
        k3 = stur_system(ti + 0.5 * dt, yi + 0.5 * dt * k2, p)
        k4 = stur_system(ti + dt, yi + dt * k3, p)
        y[:, i + 1] = yi + (dt / 6.0) * (k1 + 2 * k2 + 2 * k3 + k4)
        if y[0, i + 1] < p.eps:
            y[0, i + 1] = p.eps

    a = y[0]
    R = y[1]
    Rdot = y[2]
    H = np.gradient(np.log(np.maximum(a, p.eps)), t)
    rho = np.array([rho_R(R[i], Rdot[i], p) for i in range(len(t))])
    rhoe = np.array([rho_eff(rho[i], p) for i in range(len(t))])
    idx_b = int(np.argmin(np.abs(H)))
    t_b = float(t[idx_b])

    return {"t": t, "a": a, "H": H, "R": R, "Rdot": Rdot,
            "rho_R": rho, "rho_eff": rhoe,
            "t_bounce": t_b, "idx_bounce": idx_b}
