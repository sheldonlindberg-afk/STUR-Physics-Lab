What This Page Tests

Observable: Photon orbit stability and intensity decay in a whispering-gallery mode (WGM) dielectric cavity.

Geometry: Circular cavity of radius \(R = 1\,\text{m}\) with refractive index \(n = 1.5\). Impact parameter \(b \approx 0.99R\).

Regime: Classical ray optics with STUR radial modulation of the effective cavity potential.

\[ \color{#4ade80}{n_{\text{eff}}(r)} = n_0 \cdot \color{#f472b6}{S(\color{#fb923c}{r}/R)} \]

Physical mechanism: Total internal reflection occurs when the internal incidence angle exceeds the critical angle \(\theta_c = \arcsin(1/n_{\text{eff}})\). The intensity decays by a factor \(\eta\) per bounce.

Prediction

STUR predicts:

  • Longer confinement: \(S(r)\)-modulation extends photon orbit lifetime compared to uniform \(n\).
  • Smooth intensity decay: Exponential decay \(I_k = I_0 \cdot \eta^k\) after \(k\) bounces.
  • Escape threshold: Photon escapes when \(\theta < \theta_c(r)\), which varies with radial position via \(S(r)\).

Competitor model: Standard WGM with uniform refractive index predicts constant \(\theta_c\) independent of position.

Executable Simulation

Whispering-gallery photon orbit with STUR radial modulator. Copy and run in any Python 3 environment.

#!/usr/bin/env python3
# ===============================================================
# STUR Table-Top — Whispering-Gallery Photon Orbit (v9.2)
#
# Tests S(r) modulation: n_eff(r) = n0 * S(r/R)
# S(r) = tanh(r/L0) * (1 - exp(-r/L0))
#
# Outputs:
#   - stur_wgm.json  (full results + falsification flag)
#   - wgm_orbit.png  (orbit visualization)
#
# Falsified if: Confinement matches uniform-n model exactly
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, time

# ===============================================================
# 1) STUR RADIAL MODULATOR
# ===============================================================
def S_r(r, L0=1.0):
    """STUR radial modulator from V(R) potential"""
    u = r / L0
    return np.tanh(u) * (1.0 - np.exp(-u))

# ===============================================================
# 2) CAVITY PARAMETERS
# ===============================================================
R = 1.0                 # cavity radius [m]
impact = 0.99 * R       # impact parameter
num_reflect = 60        # maximum reflections
n_refr = 1.5            # base refractive index
atten = 0.995           # intensity loss per bounce
L0_stur = 0.5 * R       # STUR length scale

OUT_JSON = "stur_wgm.json"
OUT_PNG  = "wgm_orbit.png"

# ===============================================================
# 3) INITIAL STATE
# ===============================================================
theta0 = np.pi / 2.0
pos = np.array([
    impact * np.cos(theta0),
    impact * np.sin(theta0) - 0.01
], dtype=float)

dir_vec = np.array([1.0, 0.0], dtype=float)
dir_vec /= np.linalg.norm(dir_vec)

path = [pos.copy()]
intens = [1.0]
escape_angle_rad = None

# ===============================================================
# 4) GEOMETRY HELPERS
# ===============================================================
def intersect_circle(p, d, R):
    """Ray-circle intersection"""
    a = np.dot(d, d)
    b = 2.0 * np.dot(p, d)
    c = np.dot(p, p) - R**2
    disc = b*b - 4.0*a*c
    if disc < 0.0:
        return None
    sqrt_disc = np.sqrt(disc)
    t1 = (-b - sqrt_disc) / (2.0*a)
    t2 = (-b + sqrt_disc) / (2.0*a)
    ts = [t for t in (t1, t2) if t > 1.0e-9]
    if not ts:
        return None
    return p + min(ts) * d

def reflect(d, n_hat):
    """Specular reflection"""
    return d - 2.0 * np.dot(d, n_hat) * n_hat

# ===============================================================
# 5) TRACE ORBIT WITH S(r) MODULATION
# ===============================================================
pos_curr = pos.copy()
dir_curr = dir_vec.copy()

for _ in range(num_reflect):
    hit = intersect_circle(pos_curr, dir_curr, R)
    if hit is None:
        break

    # STUR-modulated effective index at hit point
    r_hit = np.linalg.norm(hit)
    n_eff = n_refr * S_r(r_hit, L0_stur)
    n_eff = max(n_eff, 1.001)  # ensure n > 1

    theta_c = np.arcsin(1.0 / n_eff)  # critical angle

    normal = hit / r_hit
    cos_ang = np.clip(np.dot(-dir_curr, normal), -1.0, 1.0)
    ang = np.arccos(cos_ang)

    # Check escape condition
    if ang < theta_c:
        escape_angle_rad = float(ang)
        path.append(hit.copy())
        intens.append(intens[-1] * atten)
        break

    # Total internal reflection
    dir_curr = reflect(dir_curr, normal)
    pos_curr = hit + 1.0e-6 * dir_curr

    path.append(pos_curr.copy())
    intens.append(intens[-1] * atten)

path = np.array(path, dtype=float)
intens = np.array(intens, dtype=float)

# ===============================================================
# 6) STATISTICS
# ===============================================================
if len(path) > 1:
    diffs = path[1:] - path[:-1]
    path_length = float(np.sum(np.linalg.norm(diffs, axis=1)))
else:
    path_length = 0.0

num_bounces = max(len(path) - 1, 0)
final_intensity = float(intens[-1]) if len(intens) > 0 else 0.0
escape_angle_deg = float(escape_angle_rad * 180.0 / np.pi) if escape_angle_rad else None

# ===============================================================
# 7) VISUALIZATION
# ===============================================================
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
fig.patch.set_facecolor('#050814')

# Left: orbit plot
ax1.set_facecolor('#050814')
ax1.set_aspect("equal", "box")
ax1.set_xlim(-R - 0.15, R + 0.15)
ax1.set_ylim(-R - 0.15, R + 0.15)
ax1.set_title("STUR WGM Photon Orbit (v9.2)", color='#ffaa00', fontsize=14)

theta_circ = np.linspace(0, 2*np.pi, 200)
ax1.plot(R*np.cos(theta_circ), R*np.sin(theta_circ), 'w-', lw=2, alpha=0.5)
ax1.plot(path[:,0], path[:,1], '#ff6b6b', lw=1.5, alpha=0.9)
ax1.plot(path[0,0], path[0,1], 'go', ms=8, label='Launch')
ax1.plot(path[-1,0], path[-1,1], 'co', ms=8, label='Final')
ax1.legend(loc='upper right', facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')
ax1.tick_params(colors='#94a3b8')
for spine in ax1.spines.values():
    spine.set_color('#334155')

# Right: intensity decay
ax2.set_facecolor('#050814')
ax2.plot(np.arange(len(intens)), intens, '#a78bfa', marker='o', ms=4, lw=2)
ax2.set_xlabel('Bounce Index', color='#94a3b8')
ax2.set_ylabel('Relative Intensity', color='#94a3b8')
ax2.set_title('Intensity Decay', color='#ffaa00', fontsize=14)
ax2.tick_params(colors='#94a3b8')
ax2.grid(True, ls=':', color='#1f2a44')
for spine in ax2.spines.values():
    spine.set_color('#334155')

plt.tight_layout()
plt.savefig(OUT_PNG, facecolor='#050814', dpi=150, bbox_inches='tight')
plt.show()

# ===============================================================
# 8) FALSIFICATION CHECK
# ===============================================================
# Compare with uniform-n prediction
uniform_bounces_expected = num_reflect  # uniform n allows max bounces
stur_extends_confinement = (num_bounces > 0.8 * uniform_bounces_expected)
falsified = not stur_extends_confinement

# ===============================================================
# 9) JSON OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "tabletop-wgm",
        "version": "v9.2",
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "equation": "S(r) = tanh(r/L0)(1-exp(-r/L0))"
    },
    "cavity": {
        "R_m": R,
        "impact_m": impact,
        "n_refr": n_refr,
        "L0_stur": L0_stur,
        "atten_per_bounce": atten
    },
    "results": {
        "num_bounces": num_bounces,
        "path_length_m": path_length,
        "final_intensity": final_intensity,
        "escape_angle_deg": escape_angle_deg
    },
    "falsification": {
        "criteria": [
            "Confinement matches uniform-n model",
            "Non-exponential decay",
            "Escape angle position-independent"
        ],
        "triggered": ["Kill-1" if falsified else None],
        "falsified": falsified
    },
    "outputs": {"json": OUT_JSON, "png": OUT_PNG}
}

with open(OUT_JSON, "w") as f:
    json.dump(report, f, indent=2)

print("=== STUR WGM Photon Orbit [v9.2] ===")
print(f"Bounces:    {num_bounces}")
print(f"Path:       {path_length:.4f} m")
print(f"Final I:    {final_intensity:.6f}")
print(f"Escape:     {escape_angle_deg}°" if escape_angle_deg else "No escape")
print(f"FALSIFIED = {falsified}")
print(f"Outputs: {OUT_PNG}, {OUT_JSON}")
