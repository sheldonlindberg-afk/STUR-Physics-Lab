#!/usr/bin/env python3
# ===============================================================
# STUR Resistance Corridor
#
# Complete EFT Framework: 4 fundamental inputs (M_Planck, v_EW, m_t, alpha_em), geometric framework
#
# Derives access probability from S(r) modulated forces
# Shannon entropy quantifies information content
#
# Outputs:
#   - stur_gate.json  (success probabilities and entropy)
#   - stur_gate.png   (force profile and reliability plots)
#
# Derived from Master Action; M_Planck sets the length/time scale used here
# Application layer: corridor geometry is scenario-specific
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, time

plt.style.use("dark_background")
np.random.seed(42)

OUT_JSON = "stur_gate.json"
OUT_PNG  = "stur_gate.png"

# ===============================================================
# 1) CONSTANTS
# ===============================================================
Ep, fR, tau, eta = 2.5e3, 10.0, 0.1, 0.15
E_arc = Ep * fR * tau * eta
E_pack = 1.0e6
J0 = E_pack + E_arc

T0, heat = 300.0, 50.0
cf = 1.0e4
fb = 5.0
w, phi = 2.0 * np.pi * fb, 0.6 * np.pi

ke, G = 8.9875e9, 6.6743e-11
q, m = 1.602e-19, 1.673e-27
a, sig = 1.0e-10, 1.0e10
lam = 1.0e-9
L0, el = 1.0e-9, 1.0e-9
hb, c = 1.055e-34, 3.0e8
eps = 1.0e-12
Ssp = hb / 2.0
g_sc = 1.0e-15

# ===============================================================
# 2) FORCE FUNCTIONS
# ===============================================================
S = lambda r: np.tanh(r / L0) * (1.0 - np.exp(-r / L0))
F_EM = lambda r: ke * q * q / (r * r + eps)
F_G = lambda r: -G * m * m / (r * r + eps)
F_S = lambda r: a * ((1.0 - np.exp(-r / lam)) / (r * r + eps) + sig * r * np.exp(-r / el))
F_sp = lambda r: Ssp * np.exp(-r / el)
F_SC0 = lambda r: g_sc * np.tanh(15.0 * r / L0) * (1.0 - np.exp(-r / L0))**2 * np.exp(-r / el)
F_b0 = lambda r: (2.0e-8 + 1.0e-9 * np.sin(r / 5.0)) * np.exp(-r / 40.0) / (r + eps)
F_v = lambda r: hb * c / (r**4 + eps) * 1.0e-30 * np.exp(-r / 10.0)
F_d = lambda r: 1.0e-9 / (r + eps)

def F_tot(r, t, A_b, A_s):
    F_b = F_b0(r) * (1.0 + A_b * np.sin(w * t))
    F_SC = F_SC0(r) * (1.0 - A_s * np.sin(w * t + phi))
    return S(r) * (F_EM(r) + F_G(r) + F_S(r)) + F_sp(r) + F_SC + F_v(r) + F_d(r) + F_b

def path_cost(rg, t, A_b, A_s):
    Fv = np.array([F_tot(r, t, A_b, A_s) for r in rg])
    edge_w = 0.5 * (np.abs(Fv[:-1]) + np.abs(Fv[1:]))
    return float(np.sum(edge_w))

def launch(rg, t, div, A_b, A_s, J, noise=0.2, n=200):
    c = path_cost(rg, t, A_b, A_s)
    sc = J / div
    Eb = np.random.uniform(5.0e-8, 1.0e-5, n) * sc
    need = c * (1.0 + np.random.normal(0.0, noise, n))
    return np.mean(Eb > need), c

def entropy(p):
    p = np.clip(p, 1.0e-12, 1.0 - 1.0e-12)
    return -p * np.log2(p) - (1.0 - p) * np.log2(1.0 - p)

# ===============================================================
# 3) SIMULATION
# ===============================================================
r = np.linspace(10.0, 50.0, 80)
tS = [0.0, 0.1, 0.2]
div = 1.0e7
A_s = 0.25

T = T0 + (np.random.rand() - 0.5) * heat
A_b = np.exp(-T / T0) * (0.4 + 0.3 * eta)
J = J0

noise_levels = [0.2, 0.05, 0.01]
resA, resM = [], []

for nz in noise_levels:
    sA = [launch(r, ts, div, A_b, A_s, J, noise=nz)[0] for ts in tS]
    resA.append((nz, np.mean(sA)))
    resM.append((nz, max(sA)))

nz_best, s_best = max(resM, key=lambda x: x[1])
Sinfo = entropy(s_best)

# ===============================================================
# 4) VISUALIZATION
# ===============================================================
fig, axes = plt.subplots(1, 3, figsize=(14, 4), dpi=150)
fig.patch.set_facecolor('#050814')

# Force profile
ax1 = axes[0]
ax1.set_facecolor('#050814')
r_plot = np.linspace(10.0, 50.0, 200)
F_t0 = np.array([abs(F_tot(rv, 0.0, A_b, A_s)) for rv in r_plot])
F_t1 = np.array([abs(F_tot(rv, 0.2, A_b, A_s)) for rv in r_plot])
ax1.plot(r_plot, F_t0, color='#4ade80', lw=2, label='t=0.0')
ax1.plot(r_plot, F_t1, '--', color='#60a5fa', lw=2, label='t=0.2')
ax1.set_xlabel('r (arb. units)', color='#94a3b8')
ax1.set_ylabel('|F_tot|', color='#94a3b8')
ax1.set_title('Force Profile', color='#ffaa00')
ax1.tick_params(colors='#94a3b8')
ax1.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white', fontsize=8)
ax1.grid(alpha=0.3, color='#1f2a44')
for spine in ax1.spines.values():
    spine.set_color('#334155')

# Success vs phase
ax2 = axes[1]
ax2.set_facecolor('#050814')
for nz in noise_levels:
    vals = [launch(r, ts, div, A_b, A_s, J, noise=nz)[0] * 100 for ts in tS]
    ax2.plot(tS, vals, 'o-', label=f'noise={nz}')
ax2.set_xlabel('Control phase t (s)', color='#94a3b8')
ax2.set_ylabel('Success (%)', color='#94a3b8')
ax2.set_title('Access vs Phase', color='#ffaa00')
ax2.tick_params(colors='#94a3b8')
ax2.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white', fontsize=8)
ax2.grid(alpha=0.3, color='#1f2a44')
for spine in ax2.spines.values():
    spine.set_color('#334155')

# Reliability vs noise
ax3 = axes[2]
ax3.set_facecolor('#050814')
noise_arr = np.array([x[0] for x in resA])
avg_succ = np.array([x[1] * 100 for x in resA])
max_succ = np.array([x[1] * 100 for x in resM])
ax3.plot(noise_arr, avg_succ, 'o-', color='#f472b6', label='avg')
ax3.plot(noise_arr, max_succ, 's--', color='#fbbf24', label='max')
ax3.set_xlabel('Noise level', color='#94a3b8')
ax3.set_ylabel('Success (%)', color='#94a3b8')
ax3.set_title('Reliability vs Noise', color='#ffaa00')
ax3.tick_params(colors='#94a3b8')
ax3.legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white', fontsize=8)
ax3.grid(alpha=0.3, color='#1f2a44')
for spine in ax3.spines.values():
    spine.set_color('#334155')

plt.tight_layout()
plt.savefig(OUT_PNG, facecolor='#050814', dpi=150, bbox_inches='tight')
plt.show()

# ===============================================================
# 5) JSON OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "resistance-corridor",
        "version": "4.0",
        "framework": "Complete EFT: 4 fundamental inputs (M_Planck, v_EW, m_t, alpha_em), ∞₃ helix geometry",
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "equation": "F_tot = S(r)[F_EM + F_G + F_S] + F_sp + F_SC + F_v + F_d + F_b"
    },
    "params": {
        "E_pack_J": float(E_pack),
        "T_K": float(T),
        "A_b": float(A_b),
        "A_s": float(A_s),
        "divider": float(div)
    },
    "results": {
        "noise_levels": [float(x[0]) for x in resA],
        "avg_success_percent": [float(x[1] * 100) for x in resA],
        "max_success_percent": [float(x[1] * 100) for x in resM],
        "best_success_percent": float(s_best * 100),
        "entropy_bits": float(Sinfo)
    },
    "outputs": {"json": OUT_JSON, "png": OUT_PNG}
}

with open(OUT_JSON, "w") as f:
    json.dump(report, f, indent=2)

print("=== STUR Resistance Corridor ===")
print("Complete EFT Framework: 4 fundamental inputs (M_Planck, v_EW, m_t, alpha_em), geometric framework")
print(f"Best success: {s_best*100:.2f}%")
print(f"Entropy:      {Sinfo:.3f} bits")
print(f"Outputs: {OUT_PNG}, {OUT_JSON}")
