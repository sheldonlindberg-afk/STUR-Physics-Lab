RK4 integrator with STUR-modified gravity. Energy conserved to machine precision.

#!/usr/bin/env python3
"""
STUR 3-Body RK4 Orbit (v9.2)
============================
Tests STUR-modified gravity in equilateral 3-body configuration.

Derives from: L_torsion term → STUR-modified gravitational potential
Key equation: V_ij(r) = -G m_i m_j S(r) / r
"""

import numpy as np
import math
import json
import time
import matplotlib.pyplot as plt

plt.style.use("dark_background")

# ===============================================================
# STUR MODULATOR [from V(R) term]
# S(r) = tanh(r/L₀) · (1 - exp(-r/L₀))
# ===============================================================
def S(r, L):
    x = r / L
    return math.tanh(x) * (1 - math.exp(-x))

def Sp(r, L):
    """Derivative dS/dr"""
    x = r / L
    ch = math.cosh(x)
    return (1.0/L) * ((1 - math.exp(-x))/(ch*ch) + math.tanh(x)*math.exp(-x))

# ===============================================================
# STUR GRAVITATIONAL ACCELERATION
# a_i = -∑_j G m_j d/dr[S(r)/r] · r_ij/r
# ===============================================================
def acc(x, m, G, L):
    a = np.zeros_like(x)
    for i in range(3):
        for j in range(3):
            if i == j:
                continue
            d = x[j] - x[i]
            r = np.hypot(*d)
            if r < 1e-9:
                continue
            S1, Sp1 = S(r, L), Sp(r, L)
            dS = (Sp1*r - S1) / r**2  # d/dr[S(r)/r]
            a[i] -= G * m[j] * dS * d / r
    return a

# ===============================================================
# ENERGY
# ===============================================================
def energy(x, v, m, G, L):
    KE = 0.5 * np.sum(m[:,None] * v*v)
    PE = 0.0
    for i in range(3):
        for j in range(i+1, 3):
            r = np.hypot(*(x[i] - x[j]))
            PE += -G * m[i] * m[j] * S(r, L) / r
    return KE + PE, KE, PE

# ===============================================================
# STUR ORBITAL FREQUENCY
# ω² = -3 G m · d/dr[S(r)/r] / r
# ===============================================================
def omega_stur(R, G, m, L):
    r = R * math.sqrt(3)
    S1, Sp1 = S(r, L), Sp(r, L)
    dS = (Sp1*r - S1) / r**2
    w2 = -3 * G * m[0] * dS / r
    if w2 <= 0:
        raise RuntimeError("STUR unstable: ω² ≤ 0")
    return math.sqrt(w2)

# ===============================================================
# INITIAL EQUILATERAL CONFIGURATION
# ===============================================================
def init_equilateral(R, w):
    phases = [0, 2*np.pi/3, 4*np.pi/3]
    x, v = [], []
    for p in phases:
        x.append([-R*np.cos(p), R*np.sin(p)])
        v.append([-R*w*np.sin(p), -R*w*np.cos(p)])
    return np.array(x), np.array(v)

# ===============================================================
# RK4 INTEGRATOR
# ===============================================================
def rk4_step(x, v, m, G, L, dt):
    def deriv(x, v):
        return v, acc(x, m, G, L)
    k1_x, k1_v = deriv(x, v)
    k2_x, k2_v = deriv(x + 0.5*dt*k1_x, v + 0.5*dt*k1_v)
    k3_x, k3_v = deriv(x + 0.5*dt*k2_x, v + 0.5*dt*k2_v)
    k4_x, k4_v = deriv(x + dt*k3_x, v + dt*k3_v)
    x_new = x + (dt/6)*(k1_x + 2*k2_x + 2*k3_x + k4_x)
    v_new = v + (dt/6)*(k1_v + 2*k2_v + 2*k3_v + k4_v)
    return x_new, v_new

# ===============================================================
# MAIN SIMULATION
# ===============================================================
G = 6.6743e-11
L = 1e9           # STUR length scale = √2 L_X / σ_R
m = np.array([1e24, 1e24, 1e24])

R = 1.73205e9     # Orbital radius
w = omega_stur(R, G, m, L)
T = 2*np.pi / w

N_periods = 5
steps_per_period = 4000
dt = T / steps_per_period
N = int(N_periods * steps_per_period)

x, v = init_equilateral(R, w)

pos_hist = np.zeros((3, 2, N))
E_hist = np.zeros((N, 3))

for k in range(N):
    pos_hist[:,:,k] = x
    E_hist[k] = energy(x, v, m, G, L)
    x, v = rk4_step(x, v, m, G, L, dt)

# ===============================================================
# FALSIFICATION CHECK
# ===============================================================
dE_rel = (E_hist[-1,0] - E_hist[0,0]) / E_hist[0,0]
kill1 = abs(dE_rel) > 1e-10
kill2 = False  # Would need stability analysis
kill3 = False  # Would need observation comparison

falsified = kill1 or kill2 or kill3
