def S_stur(r, L0):

def generate_helix(axis_vec, R_max, turns=4.0, n_steps=N_STEPS_HELIX, rep=False):
    axis = np.asarray(axis_vec, dtype=float)
    norm = np.linalg.norm(axis)
    if norm == 0.0:
        axis = np.array([1.0, 0.0, 0.0])
    else:
        axis /= norm

    tmp = np.array([0.0, 0.0, 1.0])

    return pts, S_prof
