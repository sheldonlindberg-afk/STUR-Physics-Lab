#!/usr/bin/env python3
# =====================================================================
# STUR 5D χ-COUPLED MULTI-FLUID SONOLUMINESCENCE ENGINE v5.0
# Includes:
#   - 1000 bubble Monte-Carlo per fluid
#   - All fluids (H2O, D2O, Ar, Xe, He, SF6, C2F6, CCl4, Air)
#   - STUR χ-torsion coupling
#   - Coulomb + strong-force enhancement
#   - Plasma heating, bremsstrahlung, ionization
#   - Spectra, R_min, T_max, χ-phase metrics
#   - JupyterLite-safe (NumPy + Matplotlib)
# =====================================================================

import numpy as np
import matplotlib.pyplot as plt
import json

plt.style.use("dark_background")

# =====================================================================
# 1. Mechanical / Fluid Database
# =====================================================================

fluids = {
 "H2O": {"rho":997,"gamma":1.33,"visc":1e-3,"sigma":0.072,"Pg0":1e5,"ion_E":12.1},
 "D2O": {"rho":1105,"gamma":1.33,"visc":1.25e-3,"sigma":0.072,"Pg0":1e5,"ion_E":12.1},
 "Ar":  {"rho":1.78,"gamma":5/3,"visc":2.2e-5,"sigma":0.013,"Pg0":1e5,"ion_E":15.8},
 "Xe":  {"rho":5.9,"gamma":5/3,"visc":2.3e-5,"sigma":0.012,"Pg0":1e5,"ion_E":12.3},
 "He":  {"rho":0.18,"gamma":5/3,"visc":1.9e-5,"sigma":0.072,"Pg0":1e5,"ion_E":24.6},
 "SF6": {"rho":6.17,"gamma":1.1,"visc":1.5e-5,"sigma":0.011,"Pg0":1e5,"ion_E":15.2},
 "C2F6":{"rho":1.59,"gamma":1.13,"visc":1.1e-5,"sigma":0.012,"Pg0":1e5,"ion_E":14.8},
 "CCl4":{"rho":1.58,"gamma":1.12,"visc":9e-4,"sigma":0.027,"Pg0":1e5,"ion_E":11.4},
 "Air":{"rho":1.23,"gamma":1.4,"visc":1.8e-5,"sigma":0.072,"Pg0":1e5,"ion_E":13.6}
}

# =====================================================================
# 2. STUR χ Coupling Constants
# =====================================================================

ke=8.9875517923e9
q=1.602e-19
N0=6e5
alpha_s=1e-28
lam=1.2e-10
sigma_L=1.2e-3
beta_chi=0.35
ell_chi=2e-10
eta_chi=0.12
Omega_chi=3.5e6
kappa_chi=1e-4

# =====================================================================
# Enhancement Factor G5
# =====================================================================

def G5(x, chi, a0):
    r=max(a0*x,1e-20)
    base=1+alpha_s*((1-np.exp(-r/lam))/(r*r+1e-30)+sigma_L*r)
    chi_term=1+beta_chi*np.exp(-x/ell_chi)*np.sin(chi)
    return base*chi_term

# =====================================================================
# Rayleigh–Plesset 5D χ-coupled RHS
# =====================================================================

def rp_rhs_5d(t,y,fluid,constants,noise):
    rho,gamma,mu,sigma_s,Pg0,ion_E=constants
    (Re,We,Pi_g,Pi_C0,omega,a0)=fluid

    x,v,T,ne,chi=y
    x=max(x,1e-8)
    T_eff=max(T,300)

    Pg=Pi_g*x**(-3*gamma)
    Pac=1.2*np.sin(omega*t)*(1+noise)

    a=(Pg-1-Pac
       -4/Re*v/x
       -2/We/x
       +Pi_C0*G5(x,chi,a0)/x**2
       -1.5*v*v
       +eta_chi*(chi-1))/x

    ne_clip=np.clip(ne,1e8,1e16)
    Ei=ion_E*1.602e-19
    kb=1.380649e-23

    adiab=-(3*gamma-1)*T*v/x
    visc=(4*mu/(Re*1e-6*Pg0))*v*v/(x*x+1e-20)
    plasma=1e-26*ne_clip*ne_clip*np.sqrt(T_eff)/(x**3+1e-20)
    chi_flash=kappa_chi*5e8*np.abs(np.sin(chi))*np.exp(-12*x)

    dT=(gamma-1)*(adiab+visc+plasma+chi_flash)

    Saha=1e22*np.exp(-Ei/(kb*T_eff))
    dne=(Saha-ne)/1e-6

    dchi=Omega_chi*np.cos(Omega_chi*t)+kappa_chi*(T*v)

    return np.array([v,a,dT,dne,dchi])

# =====================================================================
# RK4 Step
# =====================================================================

def rk4(t,y,dt,fluid,constants,noise):
    k1=rp_rhs_5d(t,y,fluid,constants,noise)
    k2=rp_rhs_5d(t+dt/2,y+dt*k1/2,fluid,constants,noise)
    k3=rp_rhs_5d(t+dt/2,y+dt*k2/2,fluid,constants,noise)
    k4=rp_rhs_5d(t+dt,y+dt*k3,fluid,constants,noise)
    return y+dt*(k1+2*k2+2*k3+k4)/6

# =====================================================================
# Single Trajectory
# =====================================================================

def run_traj(seed,fluid_name,steps=1200,t_cycle=0.04):
    fd=fluids[fluid_name]
    rho,gamma,mu,sigma_s,Pg0,ion_E=fd["rho"],fd["gamma"],fd["visc"],fd["sigma"],fd["Pg0"],fd["ion_E"]

    R0=4e-6; Pinf=1e5
    Re=rho*R0*np.sqrt(Pinf/rho)/mu
    We=rho*R0*Pinf/sigma_s
    Pi_g=Pg0/Pinf
    omega=2*np.pi*26e3*R0*np.sqrt(rho/Pinf)
    a0=(4*np.pi/3)**(1/3)*R0/(N0**(1/3))
    Pi_C0=(N0**(2/3))*ke*q*q/(Pinf*R0*R0)

    fluid_consts=(Re,We,Pi_g,Pi_C0,omega,a0)
    constants=(rho,gamma,mu,sigma_s,Pg0,ion_E)

    rng=np.random.default_rng(seed)
    noise=0.05*rng.normal()

    dt=t_cycle/steps
    t=0
    y=np.array([1.0,0.0,300.0,1e10,1.0])

    R,T,NE,CHI=[],[],[],[]

    for i in range(steps):
        R.append(y[0])
        T.append(y[2])
        NE.append(y[3])
        CHI.append(y[4])
        y=rk4(t,y,dt,fluid_consts,constants,noise)
        t+=dt

    return np.array(R),np.array(T),np.array(NE),np.array(CHI)

# =====================================================================
# Multi-Bubble Ensemble (1000 bubbles per fluid)
# =====================================================================

def run_multibubble(fluid_name,N=1000):
    Rmins=[]; Tmaxs=[]; Lambdas=[]; ChiAmp=[]
    for i in range(N):
        R,T,ne,chi=run_traj(i,fluid_name)
        Rmins.append(np.min(R))
        Tmaxs.append(np.max(T))
        peak=290e-9*(3000/np.max(T))**0.25
        Lambdas.append(peak)
        ChiAmp.append(np.max(np.abs(chi)))
    return np.array(Rmins),np.array(Tmaxs),np.array(Lambdas),np.array(ChiAmp)

# =====================================================================
# Run All Fluids
# =====================================================================

results={}

for name in fluids.keys():
    print("Running:",name)
    Rmins,Tmaxs,Lambdas,ChiAmp=run_multibubble(name,1000)
    results[name]={
      "Rmin":Rmins.tolist(),
      "Tmax":Tmaxs.tolist(),
      "lambda":Lambdas.tolist(),
      "chiamp":ChiAmp.tolist()
    }

print("Simulation complete.")
print(json.dumps(results,indent=2))

Derivations and Explanation

Rayleigh–Plesset base:
   d²R/dt² = (Pg - P_inf - P_acoustic)/ρR - (4μ/ρR²)dR/dt - 2σ/ρR³ - 3/2 (dR/dt)² / R

STUR Coulomb + strong correction:
   F_STUR ∝ (N^(2/3) ke q² / R²) * [1 + α_s ((1 - e^{-R/λ}) / R² + σ R)]

χ-torsion coupling:
   χ̈ + Ωχ² χ = κχ (T Ṙ) + ∂²R/∂χ²

Plasma heating:
   Q_plasma ∝ n_e² sqrt(T)

Bremsstrahlung spectrum:
   I(λ) ∝ n_e² sqrt(T) exp[-hc / (λ k T)]

STUR interpretation:
   Collapse forces curvature into χ-space, storing elastic torsion energy,
   released as a χ-flash when R → R_min, producing UV spike.
