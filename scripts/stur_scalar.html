Executable Simulation

Cosmological bounce with STUR-modified Friedmann equation. Requires scipy for ODE integration.

#!/usr/bin/env python3
# ===============================================================
# STUR Scalar Bounce Cosmology
#
# Tests STUR-modified Friedmann: H² = (ρ/3)(1-ρ/ρ_c)²
# Non-singular bounce when ρ → ρ_c
#
# Outputs:
#   - stur_bounce.json  (full results + falsification flag)
#   - stur_bounce.png   (a(t), H(t), ρ(t) evolution)
#
# Falsified if: Singularity forms (a_min=0) or no bounce
# ===============================================================

import numpy as np
import matplotlib.pyplot as plt
import json, time

try:
    from scipy.integrate import odeint
    HAS_SCIPY = True
except ImportError:
    HAS_SCIPY = False
    print("Warning: scipy not available, using simple Euler")

# ===============================================================
# 1) COSMOLOGICAL PARAMETERS (PLANCK UNITS)
# ===============================================================
rho_c = 1.0          # Critical density for bounce
V0 = 0.5             # Potential V(φ) = V₀ φ²
phi0 = 0.0           # Initial scalar field
phidot0 = 1.0        # Initial field velocity

OUT_JSON = "stur_bounce.json"
OUT_PNG  = "stur_bounce.png"

# ===============================================================
# 2) INITIAL CONDITIONS
# ===============================================================
rho_init = 0.5 * phidot0**2 + V0 * phi0**2
r_init = rho_init / rho_c
rho_eff = rho_init * (1 - r_init)**2
H_init = -np.sqrt(rho_eff / 3) if rho_eff > 0 else 0.0  # Contracting

a0 = 1.0
adot0 = H_init * a0

# ===============================================================
# 3) DYNAMICAL EQUATIONS
# ===============================================================
def equations(state, t):
    a, adot, phi, phidot = state
    if a <= 1e-10:
        return [0, 0, 0, 0]

    H = adot / a
    rho = 0.5 * phidot**2 + V0 * phi**2
    r = rho / rho_c

    if r >= 0.999:  # Near bounce
        r = 0.999

    # dH/dt from modified Friedmann
    dH_dt = -0.5 * phidot**2 * (1 - 4*r + 3*r**2)
    d_adot_dt = a * (dH_dt + H**2)

    # Klein-Gordon
    d_phidot_dt = -3*H*phidot - 2*V0*phi

    return [adot, d_adot_dt, phidot, d_phidot_dt]

# ===============================================================
# 4) INTEGRATION
# ===============================================================
t = np.linspace(-10, 10, 2000)
initial_state = [a0, adot0, phi0, phidot0]

if HAS_SCIPY:
    solution = odeint(equations, initial_state, t)
else:
    # Simple Euler fallback
    solution = np.zeros((len(t), 4))
    solution[0] = initial_state
    for i in range(1, len(t)):
        dt = t[i] - t[i-1]
        derivs = equations(solution[i-1], t[i-1])
        solution[i] = solution[i-1] + np.array(derivs) * dt

# Extract
a = np.maximum(solution[:,0], 1e-10)
H = solution[:,1] / a
phi = solution[:,2]
phidot = solution[:,3]
rho = 0.5 * phidot**2 + V0 * phi**2

# ===============================================================
# 5) ANALYSIS
# ===============================================================
a_min = float(np.min(a))
a_max = float(np.max(a))
H_min = float(np.min(H))
H_max = float(np.max(H))
rho_peak = float(np.max(rho))

# Check for bounce
H_crosses_zero = (H_min < 0) and (H_max > 0)
a_nonsingular = a_min > 0.01
rho_bounded = rho_peak <= rho_c * 1.1

# ===============================================================
# 6) FALSIFICATION CHECK
# ===============================================================
bounce_ok = H_crosses_zero and a_nonsingular and rho_bounded
falsified = not bounce_ok

# ===============================================================
# 7) VISUALIZATION
# ===============================================================
fig, axs = plt.subplots(3, 1, figsize=(10, 10), dpi=150)
fig.patch.set_facecolor('#050814')

for ax in axs:
    ax.set_facecolor('#050814')
    ax.tick_params(colors='#94a3b8')
    ax.grid(True, ls=':', color='#1f2a44', alpha=0.5)
    for spine in ax.spines.values():
        spine.set_color('#334155')

# Panel 1: Scale factor
axs[0].plot(t, a, '#22d3ee', lw=2)
axs[0].set_ylabel('Scale Factor a(t)', color='#94a3b8', fontsize=12)
axs[0].set_title('STUR Cosmological Bounce', color='#ffaa00', fontsize=14)
axs[0].axhline(a_min, ls='--', color='#f472b6', alpha=0.5, label=f'a_min={a_min:.4f}')
axs[0].legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')

# Panel 2: Hubble parameter
axs[1].plot(t, H, '#a78bfa', lw=2)
axs[1].set_ylabel('Hubble H(t)', color='#94a3b8', fontsize=12)
axs[1].axhline(0, ls='--', color='#94a3b8', alpha=0.5)
axs[1].fill_between(t, H, 0, where=(H<0), alpha=0.2, color='#f472b6', label='Contraction')
axs[1].fill_between(t, H, 0, where=(H>0), alpha=0.2, color='#4ade80', label='Expansion')
axs[1].legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')

# Panel 3: Energy density
axs[2].plot(t, rho, '#fbbf24', lw=2)
axs[2].axhline(rho_c, ls='--', color='#f472b6', alpha=0.7, label=f'ρ_c={rho_c}')
axs[2].set_ylabel('Energy Density ρ(t)', color='#94a3b8', fontsize=12)
axs[2].set_xlabel('Time t', color='#94a3b8', fontsize=12)
axs[2].legend(facecolor='#0a0f1f', edgecolor='#334155', labelcolor='white')

plt.tight_layout()
plt.savefig(OUT_PNG, facecolor='#050814', dpi=150, bbox_inches='tight')
plt.show()

# ===============================================================
# 8) JSON OUTPUT
# ===============================================================
report = {
    "meta": {
        "page": "scalar-bounce",
        "version": "",
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
        "equation": "H² = (ρ/3)(1-ρ/ρ_c)²"
    },
    "parameters": {
        "rho_c": rho_c,
        "V0": V0,
        "phi0": phi0,
        "phidot0": phidot0,
        "rho_init": float(rho_init),
        "H_init": float(H_init)
    },
    "results": {
        "a_min": a_min,
        "a_max": a_max,
        "H_min": H_min,
        "H_max": H_max,
        "rho_peak": rho_peak
    },
    "bounce_check": {
        "H_crosses_zero": bool(H_crosses_zero),
        "a_nonsingular": bool(a_nonsingular),
        "rho_bounded": bool(rho_bounded)
    },
    "falsification": {
        "criteria": [
            "a_min = 0 (singularity)",
            "H does not cross zero",
            "ρ exceeds ρ_c"
        ],
        "triggered": [
            "Kill-1" if not a_nonsingular else None,
            "Kill-2" if not H_crosses_zero else None,
            "Kill-3" if not rho_bounded else None
        ],
        "falsified": falsified
    },
    "outputs": {"json": OUT_JSON, "png": OUT_PNG}
}

with open(OUT_JSON, "w") as f:
    json.dump(report, f, indent=2)

print("=== STUR Scalar Bounce ===")
print(f"a_min:      {a_min:.6f}")
print(f"H range:    [{H_min:.4f}, {H_max:.4f}]")
print(f"ρ_peak:     {rho_peak:.4f} (ρ_c = {rho_c})")
print(f"Bounce:     {'YES' if H_crosses_zero else 'NO'}")
print(f"FALSIFIED = {falsified}")
print(f"Outputs: {OUT_PNG}, {OUT_JSON}")
